    10→ *
    11→ * المعادلة الملزمة لرصيد العميل (FR-03-02 + قرار 8 — لكل عملة على حدة):
    12→ *   رصيد = رصيد افتتاحي بعملته

    16→ *          AND ref_type IN ('invoice','installment','on_account'))
    17→ *     − Σ(invoice.due_amount حيث doc_type='sale_return' AND status='completed')
    18→ *
    19→ * رصيد المورّد مرآة كاملة:
    20→ *   رصيد = رصيد افتتاحي (دائن مستحق للمورّد)
    21→ *     + Σ(purchase.due_amount بذات شروط الآجل)
    22→ *     − Σ(cash_tx.amount حيث tx_type='payment' AND is_voided=0 AND ref_type كما أعلاه)

    26→ * - كل المبالغ Decimal من '@/utils/money' (لا Float) والتجميع يجري في JS فوق الصفوف
    27→ *   الخام (SUM في SQLite يُرجع REAL وقد يفقد الدقة — القوائم السريعة فقط تستخدم

   170→interface OpeningStorage {
   171→  balance: string;
   172→  currencyId: number | null;
   173→  rate: string | null;
   174→  date: string | null;
   175→}
   176→
   177→/**
   178→ * بناء قيم الرصيد الافتتاحي المخزنة:
   179→ * - بلا مبلغ → أصفار وNULLs (لا يظهر في المعادلة).
   180→ * - عملة الأساس → سعرها 1 دائماً (تعريف).
   181→ * - عملة أجنبية بمبلغ فعلي → السعر إلزامي (قرار 8: الأرصدة الافتتاحية تحمل currency_id + as_of_rate).
   182→ * - المبلغ يُقرَّب لمنازل عملته (اليمني 0).
   183→ */
   184→async function buildOpening(
   185→  amountRaw: string | undefined,
   186→  currencyIdRaw: number | undefined,
   187→  rateRaw: string | undefined,
   188→  dateRaw: string | undefined,
   189→): Promise<OpeningStorage> {
   190→  const db = await getDb();
   191→  const baseRows = await db.all<{ id: number }>('SELECT id FROM currency WHERE is_base = 1 LIMIT 1');
   192→  if (baseRows.length === 0) {
   193→    throw new Error('لم تُعرَّف عملة أساسية بعد — أكمل الإعداد الأولي للتطبيق أولاً');
   194→  }
   195→  const baseId = Number(baseRows[0].id);
   196→
   197→  const amount = amountRaw !== undefined && amountRaw.trim() !== '' ? amountRaw.trim() : '0';
   198→  const amt = dec(amount);
   199→
   200→  const currencyId = currencyIdRaw ?? baseId;
   201→  const cur = await db.all<{ is_base: number; decimals: number; is_active: number }>(
   202→    'SELECT is_base, decimals, is_active FROM currency WHERE id = ?',
   203→    [currencyId],
   204→  );
   205→  if (cur.length === 0) {
   206→    throw new Error(`عملة الرصيد الافتتاحي غير موجودة (رقم ${currencyId}) — اختر عملة من القائمة`);
   207→  }
   208→  if (Number(cur[0].is_active) !== 1) {
   209→    throw new Error('عملة الرصيد الافتتاحي موقوفة — اختر عملة مفعّلة أو أعد تفعيلها من شاشة العملات');
   210→  }
   211→  const isBase = Number(cur[0].is_base) === 1;
   212→
   213→  if (amt.isZero()) {
   214→    return {
   215→      balance: '0',
   216→      currencyId: currencyIdRaw !== undefined ? currencyId : null,
   217→      rate: isBase ? '1' : rateRaw !== undefined && rateRaw.trim() !== '' ? money(rateRaw) : null,
   218→      date: dateRaw ?? null,
   219→    };
   220→  }
   221→
   222→  let rate: string;
   223→  if (isBase) {
   224→    rate = '1';
   225→  } else if (rateRaw !== undefined && rateRaw.trim() !== '') {
   226→    rate = money(rateRaw);
   227→  } else {
   228→    throw new Error('سعر الصرف مطلوب للرصيد الافتتاحي بعملة غير الأساس — أدخل سعر يوم الرصيد (مثل «530»)');
   229→  }
   230→
   231→  const date = dateRaw ?? todayISO();
   232→  if (!ISO_DATE_RE.test(date)) {
   233→    throw new Error('تاريخ الرصيد الافتتاحي غير صالح — استخدم صيغة YYYY-MM-DD');
   234→  }
   235→  return { balance: money(roundTo(amt, Number(cur[0].decimals))), currencyId, rate, date };
   236→}

   269→/** إنشاء عميل (FR-03-01) — ذرّي مع قيد تدقيق. يُرجع المعرّف الجديد. */
   270→export async function createCustomer(input: PartyInput): Promise<number> {
   271→  const v = parseInput(input, 'العميل');
   272→  const opening = await buildOpening(v.openingBalance, v.openingCurrencyId, v.openingRate, v.openingDate);
   273→  const db = await getDb();
   274→  const now = new Date().toISOString();
   275→  let newId = 0;
   276→  await db.transaction(async () => {
   277→    const res = await db.run(
   278→      'INSERT INTO customer(name, phone, whatsapp, address, area, credit_limit, opening_balance, ' +
   279→        'opening_balance_currency_id, opening_balance_rate, opening_balance_date, notes, is_archived, created_at, updated_at, created_by) ' +
   280→        'VALUES(?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 0, ?, ?, ?)',
   281→      [
   282→        v.name,
   283→        v.phone && v.phone.length > 0 ? v.phone : null,
   284→        v.whatsapp && v.whatsapp.length > 0 ? v.whatsapp : null,
   285→        v.address && v.address.length > 0 ? v.address : null,
   286→        v.area && v.area.length > 0 ? v.area : null,
   287→        v.creditLimit === null ? null : v.creditLimit === undefined ? null : money(v.creditLimit),
   288→        opening.balance,
   289→        opening.currencyId,
   290→        opening.rate,
   291→        opening.date,
   292→        v.notes && v.notes.length > 0 ? v.notes : null,
   293→        now,
   294→        now,
   295→        getCurrentUserId() ?? null,
   296→      ],
   297→    );
   298→    newId = Number(res.lastInsertRowId);
   299→    await logAudit('create_customer', {
   300→      entity: 'customer',
   301→      entityId: newId,
   302→      details: {
   303→        name: v.name,
   304→        opening: opening.balance === '0' ? null : { balance: opening.balance, currencyId: opening.currencyId, rate: opening.rate, date: opening.date },
   305→        creditLimit: v.creditLimit ?? null,
   306→      },
   307→    });
   308→  });
   309→  return newId;
   310→}

   312→/** تعديل عميل — الحقول غير الممررة تبقى كما هي (Partial). */
   313→export async function updateCustomer(id: number, input: Partial<PartyInput>): Promise<void> {
   314→  const v = parseUpdate(input, 'العميل');
   315→  const db = await getDb();
   316→  const existing = await db.all<Record<string, unknown>>(`${CUSTOMER_SELECT} WHERE id = ?`, [id]);
   317→  if (existing.length === 0) {
   318→    throw new Error(`عميل غير موجود (رقم ${id}) — لا يمكن تعديله`);
   319→  }
   320→  const cur = normalizeCustomer(existing[0]);
   321→
   322→  // دمج الرصيد الافتتاحي: ما لم يُمرَّر يبقى من الصف الحالي (ثم يُعاد بناؤه بقواعده)
   323→  const opening = await buildOpening(
   324→    v.openingBalance !== undefined ? v.openingBalance : cur.openingBalance,
   325→    v.openingCurrencyId !== undefined ? v.openingCurrencyId : cur.openingCurrencyId ?? undefined,
   326→    v.openingRate !== undefined ? v.openingRate : cur.openingRate ?? undefined,
   327→    v.openingDate !== undefined ? v.openingDate : cur.openingDate ?? undefined,
   328→  );
   329→
   330→  const sets: string[] = ['updated_at = ?'];
   331→  const params: unknown[] = [new Date().toISOString()];
   332→  if (v.name !== undefined) {
   333→    sets.push('name = ?');
   334→    params.push(v.name);
   335→  }
   336→  if (v.phone !== undefined) {
   337→    sets.push('phone = ?');
   338→    params.push(v.phone.length > 0 ? v.phone : null);
   339→  }
   340→  if (v.whatsapp !== undefined) {
   341→    sets.push('whatsapp = ?');
   342→    params.push(v.whatsapp.length > 0 ? v.whatsapp : null);
   343→  }
   344→  if (v.address !== undefined) {
   345→    sets.push('address = ?');
   346→    params.push(v.address.length > 0 ? v.address : null);
   347→  }
   348→  if (v.area !== undefined) {
   349→    sets.push('area = ?');
   350→    params.push(v.area.length > 0 ? v.area : null);
   351→  }
   352→  if (v.creditLimit !== undefined) {
   353→    // undefined = لا تغيير / null = بلا حد (NULL) / نص = قيمة
   354→    sets.push('credit_limit = ?');
   355→    params.push(v.creditLimit === null ? null : money(v.creditLimit));
   356→  }
   357→  if (v.notes !== undefined) {
   358→    sets.push('notes = ?');
   359→    params.push(v.notes.length > 0 ? v.notes : null);
   360→  }
   361→  sets.push('opening_balance = ?', 'opening_balance_currency_id = ?', 'opening_balance_rate = ?', 'opening_balance_date = ?');
   362→  params.push(opening.balance, opening.currencyId, opening.rate, opening.date);
   363→
   364→  params.push(id);
   365→  await db.transaction(async () => {
   366→    await db.run(`UPDATE customer SET ${sets.join(', ')} WHERE id = ?`, params);
   367→    await logAudit('update_customer', {
   368→      entity: 'customer',
   369→      entityId: id,
   370→      details: { fields: Object.keys(v), opening: opening.balance === '0' ? null : opening },
   371→    });
   372→  });
   373→}

   375→/**
   376→ * أرشفة عميل (FR-03-09): لا حذف إطلاقاً — الطرف له حركات يُؤرشف فقط،
   377→ * وحتى بلا حركات لا يوفر الدومين مسار حذف (قرار حماية البيانات).
   378→ */
   379→export async function archiveCustomer(id: number): Promise<void> {
   380→  const db = await getDb();
   381→  const row = await db.all<{ name: string; is_archived: number }>(
   382→    'SELECT name, is_archived FROM customer WHERE id = ?',
   383→    [id],
   384→  );
   385→  if (row.length === 0) {
   386→    throw new Error(`عميل غير موجود (رقم ${id}) — لا يمكن أرشفته`);
   387→  }
   388→  if (Number(row[0].is_archived) === 1) return; // مؤرشف مسبقاً — نجاح صامت
   389→  const inv = await db.all<{ c: number }>('SELECT count(*) AS c FROM invoice WHERE customer_id = ?', [id]);
   390→  const cash = await db.all<{ c: number }>('SELECT count(*) AS c FROM cash_tx WHERE customer_id = ?', [id]);
   391→  await db.transaction(async () => {
   392→    await db.run('UPDATE customer SET is_archived = 1, updated_at = ? WHERE id = ?', [new Date().toISOString(), id]);
   393→    await logAudit('archive_customer', {
   394→      entity: 'customer',
   395→      entityId: id,
   396→      details: {
   397→        name: row[0].name,
   398→        invoices: Number(inv[0]?.c ?? 0),
   399→        cashTxs: Number(cash[0]?.c ?? 0),
   400→        rule: 'FR-03-09: أرشفة فقط — لا حذف لطرف له حركات',
   401→      },
   402→    });
   403→  });
   404→}

   434→    createdAt: (r.createdAt as string | null) ?? null,

   441→export async function createSupplier(input: SupplierInput): Promise<number> {

   450→        'opening_balance_rate, opening_balance_date, notes, is_archived, created_at, updated_at, created_by) ' +
   451→        'VALUES(?, ?, ?, ?, ?, ?, ?, ?, 0, ?, ?, ?)',

   453→        v.name,
   454→        v.phone && v.phone.length > 0 ? v.phone : null,
   455→        v.address && v.address.length > 0 ? v.address : null,
   456→        opening.balance,
   457→        opening.currencyId,
   458→        opening.rate,
   459→        opening.date,
   460→        v.notes && v.notes.length > 0 ? v.notes : null,

   463→        getCurrentUserId() ?? null,

   467→    await logAudit('create_supplier', {
   468→      entity: 'supplier',
   469→      entityId: newId,
   470→      details: {
   471→        name: v.name,
   472→        opening: opening.balance === '0' ? null : { balance: opening.balance, currencyId: opening.currencyId, rate: opening.rate, date: opening.date },

   480→export async function updateSupplier(id: number, input: Partial<SupplierInput>): Promise<void> {
   481→  const v = parseUpdate(input, 'المورّد');
   482→  const db = await getDb();
   483→  const existing = await db.all<Record<string, unknown>>(`${SUPPLIER_SELECT} WHERE id = ?`, [id]);
   484→  if (existing.length === 0) {
   485→    throw new Error(`مورّد غير موجود (رقم ${id}) — لا يمكن تعديله`);
   486→  }
   487→  const cur = normalizeSupplier(existing[0]);
   488→  const opening = await buildOpening(
   489→    v.openingBalance !== undefined ? v.openingBalance : cur.openingBalance,
   490→    v.openingCurrencyId !== undefined ? v.openingCurrencyId : cur.openingCurrencyId ?? undefined,
   491→    v.openingRate !== undefined ? v.openingRate : cur.openingRate ?? undefined,
   492→    v.openingDate !== undefined ? v.openingDate : cur.openingDate ?? undefined,
   493→  );
   494→
   495→  const sets: string[] = ['updated_at = ?'];
   496→  const params: unknown[] = [new Date().toISOString()];
   497→  if (v.name !== undefined) {
   498→    sets.push('name = ?');
   499→    params.push(v.name);
   500→  }
   501→  if (v.phone !== undefined) {
   502→    sets.push('phone = ?');
   503→    params.push(v.phone.length > 0 ? v.phone : null);
   504→  }
   505→  if (v.address !== undefined) {
   506→    sets.push('address = ?');
   507→    params.push(v.address.length > 0 ? v.address : null);
   508→  }
   509→  if (v.notes !== undefined) {
   510→    sets.push('notes = ?');
   511→    params.push(v.notes.length > 0 ? v.notes : null);
   512→  }
   513→  sets.push('opening_balance = ?', 'opening_balance_currency_id = ?', 'opening_balance_rate = ?', 'opening_balance_date = ?');
   514→  params.push(opening.balance, opening.currencyId, opening.rate, opening.date);
   515→
   516→  params.push(id);
   517→  await db.transaction(async () => {
   518→    await db.run(`UPDATE supplier SET ${sets.join(', ')} WHERE id = ?`, params);
   519→    await logAudit('update_supplier', { entity: 'supplier', entityId: id, details: { fields: Object.keys(v) } });
   520→  });
   521→}

   523→/** أرشفة مورّد — نفس قاعدة FR-03-09 (لا حذف أبداً). */
   524→export async function archiveSupplier(id: number): Promise<void> {
   525→  const db = await getDb();
   526→  const row = await db.all<{ name: string; is_archived: number }>(
   527→    'SELECT name, is_archived FROM supplier WHERE id = ?',
   528→    [id],
   529→  );
   530→  if (row.length === 0) {
   531→    throw new Error(`مورّد غير موجود (رقم ${id}) — لا يمكن أرشفته`);
   532→  }
   533→  if (Number(row[0].is_archived) === 1) return;
   534→  const inv = await db.all<{ c: number }>('SELECT count(*) AS c FROM invoice WHERE supplier_id = ?', [id]);
   535→  const cash = await db.all<{ c: number }>('SELECT count(*) AS c FROM cash_tx WHERE supplier_id = ?', [id]);
   536→  await db.transaction(async () => {
   537→    await db.run('UPDATE supplier SET is_archived = 1, updated_at = ? WHERE id = ?', [new Date().toISOString(), id]);
   538→    await logAudit('archive_supplier', {
   539→      entity: 'supplier',
   540→      entityId: id,
   541→      details: {
   542→        name: row[0].name,
   543→        invoices: Number(inv[0]?.c ?? 0),
   544→        cashTxs: Number(cash[0]?.c ?? 0),
   545→        rule: 'FR-03-09: أرشفة فقط — لا حذف لطرف له حركات',
   546→      },
   547→    });
   548→  });
   549→}

   558→// ============ المعادلة الملزمة — الأرصدة لكل عملة (FR-03-02 + قرار 8) ============
   559→
   560→interface CurrencyAgg {
   561→  opening: Decimal;
   562→  docs: Decimal;
   563→  payments: Decimal;
   564→  returns: Decimal;
   565→  hasRows: boolean;
   566→}
   567→
   568→function aggEntry(map: Map<number, CurrencyAgg>, currencyId: number): CurrencyAgg {
   569→  let e = map.get(currencyId);
   570→  if (e === undefined) {
   571→    e = { opening: dec(0), docs: dec(0), payments: dec(0), returns: dec(0), hasRows: false };
   572→    map.set(currencyId, e);
   573→  }
   574→  return e;
   575→}
   576→
   577→/**
   578→ * محرك الأرصدة المشترك (عميل/مورّد) — يجمع الصفوف الخام ويجمعها بـ Decimal
   579→ * (دقة كاملة بلا float)، لكل عملة على حدة (قرار 8).
   580→ */
   581→async function partyBalances(
   582→  kind: 'customer' | 'supplier',
   583→  partyId: number,
   584→): Promise<PartyBalance[]> {
   585→  const db = await getDb();
   586→  const table = kind === 'customer' ? 'customer' : 'supplier';
   587→  const idCol = kind === 'customer' ? 'customer_id' : 'supplier_id';
   588→  const purchaseType = kind === 'customer' ? 'sale' : 'purchase';
   589→  const returnType = kind === 'customer' ? 'sale_return' : 'purchase_return';
   590→  const cashType = kind === 'customer' ? 'receipt' : 'payment';
   591→
   592→  // 1) الصف الحالي (الرصيد الافتتاحي بعملته)
   593→  const partyRows = await db.all<{
   594→    opening_balance: string | number | null;
   595→    opening_balance_currency_id: number | null;
   596→  }>(`SELECT opening_balance, opening_balance_currency_id FROM ${table} WHERE id = ?`, [partyId]);
   597→  if (partyRows.length === 0) {
   598→    throw new Error(kind === 'customer' ? `عميل غير موجود (رقم ${partyId})` : `مورّد غير موجود (رقم ${partyId})`);
   599→  }
   600→
   601→  const perCurrency = new Map<number, CurrencyAgg>();
   602→  const openingAmount = dec(partyRows[0].opening_balance ?? 0);
   603→  const openingCurrencyId = partyRows[0].opening_balance_currency_id;
   604→  if (openingCurrencyId !== null && openingCurrencyId !== undefined && !openingAmount.isZero()) {
   605→    aggEntry(perCurrency, Number(openingCurrencyId)).opening = openingAmount;
   606→  }
   607→
   608→  // 2) الفواتير الآجلة والمرتجعات (الموجة 4 تملؤها — المعادلة جاهزة)
   609→  const invRows = await db.all<{ currency_id: number; doc_type: string; pay_status: string; due_amount: string | number }>(
   610→    'SELECT currency_id, doc_type, pay_status, due_amount FROM invoice ' +
   611→      `WHERE ${idCol} = ? AND status = 'completed' AND doc_type IN (?, ?)`,
   612→    [partyId, purchaseType, returnType],
   613→  );
   614→  for (const r of invRows) {
   615→    const e = aggEntry(perCurrency, Number(r.currency_id));
   616→    e.hasRows = true;
   617→    if (r.doc_type === purchaseType && (r.pay_status === 'credit' || r.pay_status === 'mixed')) {
   618→      e.docs = e.docs.plus(dec(r.due_amount));
   619→    } else if (r.doc_type === returnType) {
   620→      e.returns = e.returns.plus(dec(r.due_amount));
   621→    }
   622→  }
   623→
   624→  // 3) السندات الفعلية (غير المعكوسة) المرتبطة بالطرف
   625→  const cashRows = await db.all<{ currency_id: number; amount: string | number }>(
   626→    'SELECT currency_id, amount FROM cash_tx ' +
   627→      `WHERE ${idCol} = ? AND tx_type = ? AND is_voided = 0 ` +
   628→      "AND ref_type IN ('invoice','installment','on_account')",
   629→    [partyId, cashType],
   630→  );
   631→  for (const r of cashRows) {
   632→    const e = aggEntry(perCurrency, Number(r.currency_id));
   633→    e.hasRows = true;
   634→    e.payments = e.payments.plus(dec(r.amount));
   635→  }
   636→
   637→  // 4) تجميع العملات المعنية (ذات حركة فعلة أو رصيد افتتاحي) وقراءة رموزها
   638→  const involved = [...perCurrency.entries()]
   639→    .filter(([, e]) => e.hasRows || !e.opening.isZero())
   640→    .map(([currencyId, e]) => {
   641→      const balance = e.opening.plus(e.docs).minus(e.payments).minus(e.returns);
   642→      return { currencyId, balance };
   643→    });
   644→  if (involved.length === 0) return [];
   645→
   646→  const curRows = await db.all<{ id: number; code: string; is_base: number }>(
   647→    'SELECT id, code, is_base FROM currency',
   648→  );
   649→  const byId = new Map(curRows.map((c) => [Number(c.id), c]));
   650→
   651→  return involved
   652→    .filter((x) => byId.has(x.currencyId))
   653→    .sort((a, b) => {
   654→      const ba = Number(byId.get(a.currencyId)?.is_base ?? 0);
   655→      const bb = Number(byId.get(b.currencyId)?.is_base ?? 0);
   656→      if (ba !== bb) return bb - ba; // الأساس أولاً
   657→      return (byId.get(a.currencyId)?.code ?? '').localeCompare(byId.get(b.currencyId)?.code ?? '');
   658→    })
   659→    .map((x) => ({
   660→      currencyId: x.currencyId,
   661→      code: byId.get(x.currencyId)?.code ?? String(x.currencyId),
   662→      balance: money(x.balance),
   663→    }));
   664→}
   665→
   666→/** أرصدة العميل لكل عملة على حدة — موجب = مدين عليه، سالب = دائن لصالحه (قرار 8). */
   667→export async function customerBalances(customerId: number): Promise<PartyBalance[]> {
   668→  return partyBalances('customer', customerId);
   669→}

   671→/** أرصدة المورّد لكل عملة على حدة — موجب = مستحق له (علينا)، سالب = دائن لصالحنا. */
   672→export async function supplierBalances(supplierId: number): Promise<PartyBalance[]> {
   673→  return partyBalances('supplier', supplierId);

   678→/** تهريب محارف LIKE الخاصة (% _ \). */
   679→function likeEscape(q: string): string {
   680→  return q.replace(/[\\%_]/g, (ch) => `\\${ch}`);
   681→}
   682→
   683→/**
   684→ * رصيد الطرف بعملة الأساس — حساب مبسّط بعرضة واحدة (المطلوب للقائمة فقط):
   685→ * مكونات عملة الأساس حصراً + الرصيد الافتتاحي (بسعره المسجل إن كان بعملة أخرى).
   686→ * أرصدة العملات الأخرى تُعرض مفصولة في ملف الطرف (قرار 8 — العرض المجمّع V1.1).
   687→ * SUM في SQLite يُرجع REAL — نقرّب لمنازل NUMERIC(14,4) ثم نطبع نصاً.
   688→ */
   689→async function searchParties(
   690→  kind: 'customer' | 'supplier',
   691→  q: string,
   692→): Promise<CustomerListRow[]> {
   693→  const db = await getDb();
   694→  const base = await db.all<{ id: number }>('SELECT id FROM currency WHERE is_base = 1 LIMIT 1');
   695→  if (base.length === 0) {
   696→    throw new Error('لم تُعرَّف عملة أساسية بعد — أكمل الإعداد الأولي للتطبيق أولاً');
   697→  }
   698→  const baseId = Number(base[0].id);
   699→
   700→  const table = kind === 'customer' ? 'customer' : 'supplier';
   701→  const idCol = kind === 'customer' ? 'customer_id' : 'supplier_id';
   702→  const purchaseType = kind === 'customer' ? 'sale' : 'purchase';
   703→  const returnType = kind === 'customer' ? 'sale_return' : 'purchase_return';
   704→  const cashType = kind === 'customer' ? 'receipt' : 'payment';
   705→
   706→  const search = q.trim();
   707→  const where =
   708→    search.length > 0 ? " AND (p.name LIKE ? ESCAPE '\\' OR IFNULL(p.phone, '') LIKE ? ESCAPE '\\')" : '';
   709→  // ترتيب المعاملات: عملة الأساس في CASE الافتتاحي، ثم (doc_type للآجل، العملة)،
   710→  // (tx_type للسند، العملة)، (doc_type للمرتجع، العملة)، ثم معاملا البحث.
   711→  const params: unknown[] = [
   712→    baseId,
   713→    purchaseType,
   714→    baseId,
   715→    cashType,
   716→    baseId,
   717→    returnType,
   718→    baseId,
   719→  ];
   720→  if (search.length > 0) {
   721→    const esc = `%${likeEscape(search)}%`;
   722→    params.push(esc, esc);
   723→  }
   724→
   725→  const rows = await db.all<{ id: number; name: string; phone: string | null; area: string | null; base_balance: string | number }>(
   726→    'SELECT p.id, p.name, p.phone' +
   727→      (kind === 'customer' ? ', p.area' : ', NULL AS area') +
   728→      ', CAST(ROUND(' +
   729→      'CASE WHEN p.opening_balance_currency_id = ? THEN IFNULL(p.opening_balance, 0)' +
   730→      '     WHEN p.opening_balance_currency_id IS NOT NULL AND p.opening_balance_rate IS NOT NULL' +
   731→      '          AND IFNULL(p.opening_balance, 0) <> 0 THEN p.opening_balance * p.opening_balance_rate' +
   732→      '     ELSE 0 END' +
   733→      ` + IFNULL((SELECT SUM(i.due_amount) FROM invoice i WHERE i.${idCol} = p.id AND i.doc_type = ? AND i.status = 'completed' AND i.pay_status IN ('credit','mixed') AND i.currency_id = ?), 0)` +
   734→      ` - IFNULL((SELECT SUM(t.amount) FROM cash_tx t WHERE t.${idCol} = p.id AND t.tx_type = ? AND t.is_voided = 0 AND t.ref_type IN ('invoice','installment','on_account') AND t.currency_id = ?), 0)` +
   735→      ` - IFNULL((SELECT SUM(r.due_amount) FROM invoice r WHERE r.${idCol} = p.id AND r.doc_type = ? AND r.status = 'completed' AND r.currency_id = ?), 0)` +
   736→      `, 4) AS TEXT) AS base_balance ` +
   737→      `FROM ${table} p WHERE p.is_archived = 0${where} ORDER BY p.name ASC, p.id ASC LIMIT 300`,
   738→    params,
   739→  );
   740→
   741→  return rows.map((r) => ({
   742→    id: Number(r.id),
   743→    name: String(r.name),
   744→    phone: (r.phone as string | null) ?? null,
   745→    area: (r.area as string | null) ?? null,
   746→    baseBalance: money(dec(r.base_balance)),
   747→  }));
   748→}
