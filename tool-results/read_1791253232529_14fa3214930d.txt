   223→   223→async function loadProducts(db: DbEngine, items: SaleItemInput[]): Promise<Map<number, ProductLite>> {
   224→   224→  const ids = [...new Set(items.filter((it) => it.productId !== null).map((it) => it.productId as number))];
   225→   225→  const products = new Map<number, ProductLite>();
   226→   226→  for (const id of ids) {
   227→   227→    const rows = await db.all<{ id: number; name: string; cost_price: string | number | null; is_service: number; is_archived: number }>(
   228→   228→      'SELECT id, name, cost_price, is_service, is_archived FROM product WHERE id = ?',
   229→   229→      [id],
   230→   230→    );
   231→   231→    const r = rows[0];
   232→   232→    if (r === undefined) {
   233→   233→      throw new Error(`صنف غير موجود (رقم ${id}) — ربما حُذف رابطه، أعد إضافة الصنف للفاتورة`);
   234→   234→    }
   235→   235→    if (Number(r.is_archived) === 1) {
   236→   236→      throw new Error(`الصنف «${r.name}» مؤرشف ولا يمكن بيعه — أعد تفعيله من شاشة المخزون أولاً`);
   237→   237→    }
   238→   238→    products.set(Number(r.id), {
   239→   239→      id: Number(r.id),
   240→   240→      name: r.name,
   241→   241→      costPrice: dec(r.cost_price ?? 0),
   242→   242→      isService: Number(r.is_service) === 1,
   243→   243→    });
   244→   244→  }
   245→   245→  return products;
   246→   246→}

   248→   248→/** بناء السياق: تحقق + حسابات + سعر صرف + فترات + دفع + حد ائتمان (بلا كتابة). */
   249→   249→async function buildSaleCtx(db: DbEngine, v: SaveSaleParsed, opts: { fromDraft?: boolean }): Promise<SaleCtx> {
   250→   250→  const now = new Date().toISOString();
   251→   251→  const issuedAt = v.issuedAt ?? todayISO();
   252→   252→  const isDraft = v.saveAsDraft === true;
   253→   253→
   254→   254→  // 1) المراجع: مستودع/عملة (+ صندوق/عميل عند الحاجة)
   255→   255→  const wh = await db.all<{ id: number; name: string }>(
   256→   256→    'SELECT id, name FROM warehouse WHERE id = ? AND is_archived = 0',
   257→   257→    [v.warehouseId],
   258→   258→  );
   259→   259→  if (wh.length === 0) {
   260→   260→    throw new Error(`المستودع (رقم ${v.warehouseId}) غير موجود أو مؤرشف — اختر مستودعاً صالحاً`);
   261→   261→  }
   262→   262→  const cur = await db.all<{ id: number; code: string; is_active: number }>(
   263→   263→    'SELECT id, code, is_active FROM currency WHERE id = ?',
   264→   264→    [v.currencyId],
   265→   265→  );
   266→   266→  if (cur.length === 0 || Number(cur[0]!.is_active) !== 1) {
   267→   267→    throw new Error(`العملة (رقم ${v.currencyId}) غير موجودة أو موقوفة — اختر عملة مفعلة`);
   268→   268→  }
   269→   269→
   270→   270→  // 2) الأصناف والخدمات
   271→   271→  const products = await loadProducts(db, v.items);
   272→   272→  for (const it of v.items) {
   273→   273→    if (it.productId === null && (it.lineDesc ?? '').trim().length === 0) {
   274→   274→      throw new Error('سطر الخدمة الحرة يتطلب وصفاً — أدخل وصف البند (مثل «توصيل») أو اربطه بصنف');
   275→   275→    }
   276→   276→  }
   277→   277→
   278→   278→  // 3) الحسابات (computeInvoiceTotals فقط — لا حساب يدوي)
   279→   279→  const taxModeRaw = await getSetting('invoicing.tax_mode');
   280→   280→  const taxMode: 'per_item' | 'on_total' = taxModeRaw === 'per_item' ? 'per_item' : 'on_total';
   281→   281→  const companyRows = await db.all<{ tax_rate: string | number | null }>('SELECT tax_rate FROM company LIMIT 1');
   282→   282→  const taxRate = money(companyRows[0]?.tax_rate ?? 0);
   283→   283→
   284→   284→  const lines: InvoiceLineInput[] = v.items.map((it) => {
   285→   285→    const p = it.productId !== null ? products.get(it.productId) : undefined;
   286→   286→    const lineCost = p !== undefined && !p.isService ? money(roundTo(p.costPrice.times(dec(it.qty)), 4)) : '0';
   287→   287→    return {
   288→   288→      productId: it.productId,
   289→   289→      lineDesc: it.lineDesc,
   290→   290→      qty: it.qty,
   291→   291→      unitPrice: it.unitPrice,
   292→   292→      discountPercent: it.discountPercent ?? '0',
   293→   293→      taxPercent: it.taxPercent ?? '0',
   294→   294→      lineCost,
   295→   295→    };
   296→   296→  });
   297→   297→  const totals = computeInvoiceTotals(lines, { taxMode, taxRate, invoiceDiscount: v.invoiceDiscount ?? '0' });
   298→   298→  assertTotalsValid(totals); // منع صافي ≤ 0 (FR-02-05)
   299→   299→
   300→   300→  // 4) سعر الصرف (قرار 3): MissingRateError تمر كما هي — قبل أي كتابة
   301→   301→  const base = await getBaseCurrency();
   302→   302→  const snapshot = v.currencyId === base.id ? { rate: '1', rateIsFallback: false } : await getRateSnapshot(v.currencyId, issuedAt);
   303→   303→
   304→   304→  // 5) الفترات والتأريخ (قاعدة 5.4-11 + قرار 6) — assertBackdateAllowed يقيّد backdate في audit
   305→   305→  await assertPeriodOpen(issuedAt);
   306→   306→  await assertBackdateAllowed(issuedAt, { managerConfirmed: v.managerConfirmedBackdate });
   307→   307→
   308→   308→  const total = dec(totals.total);
   309→   309→
   310→   310→  // 6) تقسيم الدفع (المسودة: نوايا بلا متطلبات)
   311→   311→  let paid = dec(0);
   312→   312→  let due = dec(0);
   313→   313→  if (isDraft) {
   314→   314→    due = total;
   315→   315→  } else if (v.payType === 'cash') {
   316→   316→    paid = total;
   317→   317→  } else if (v.payType === 'credit') {
   318→   318→    due = total;
   319→   319→  } else {
   320→   320→    if (v.cashPart === undefined || v.cashPart === '') {
   321→   321→      throw new Error('الفاتورة المختلطة تتطلب مبلغ الجزء النقدي — أدخل المبلغ المدفوع نقداً أو اختر نوع دفع آخر');
   322→   322→    }
   323→   323→    const cp = dec(v.cashPart);
   324→   324→    if (cp.lessThanOrEqualTo(0)) {
   325→   325→      throw new Error('الجزء النقدي في الفاتورة المختلطة يجب أن يكون أكبر من صفر — للآجل الكامل استخدم «حفظ آجل»');
   326→   326→    }
   327→   327→    if (cp.greaterThan(total)) {
   328→   328→      throw new Error('الجزء النقدي أكبر من إجمالي الفاتورة — خفّض المبلغ أو استخدم الدفع النقدي الكامل');
   329→   329→    }
   330→   330→    paid = cp;
   331→   331→    due = total.minus(cp);
   332→   332→  }
   333→   333→
   334→   334→  // 7) متطلبات الطرف/الصندوق (للمكتملة فقط)
   335→   335→  let customerId: number | null = v.customerId ?? null;
   336→   336→  let cashboxId: number | null = v.cashboxId ?? null;
   337→   337→  if (!isDraft) {
   338→   338→    if ((v.payType === 'credit' || v.payType === 'mixed') && customerId === null) {
   339→   339→      throw new Error('حفظ الفاتورة الآجلة يتطلب اختيار عميل أولاً — اختر العميل من شريط الأعلى أو استخدم الدفع النقدي');
   340→   340→    }
   341→   341→    if ((v.payType === 'cash' || v.payType === 'mixed') && cashboxId === null) {
   342→   342→      throw new Error('الدفع النقدي يتطلب اختيار صندوق — اختر الصندوق من شريط الأعلى قبل الحفظ');
   343→   343→    }
   344→   344→  }
   345→   345→  if (customerId !== null) {
   346→   346→    const cust = await db.all<{ id: number; name: string; is_archived: number }>(
   347→   347→      'SELECT id, name, is_archived FROM customer WHERE id = ?',
   348→   348→      [customerId],
   349→   349→    );
   350→   350→    if (cust.length === 0) {
   351→   351→      throw new Error(`عميل غير موجود (رقم ${customerId}) — أعد اختيار العميل من القائمة`);
   352→   352→    }
   353→   353→    if (Number(cust[0]!.is_archived) === 1) {
   354→   354→      throw new Error(`العميل «${cust[0]!.name}» مؤرشف — أعد تفعيله من شاشة الأطراف أولاً`);
   355→   355→    }
   356→   356→  }
   357→   357→  if (cashboxId !== null) {
   358→   358→    const box = await db.all<{ id: number; name: string; is_archived: number }>(
   359→   359→      'SELECT id, name, is_archived FROM cashbox WHERE id = ?',
   360→   360→      [cashboxId],
   361→   361→    );
   362→   362→    if (box.length === 0) {
   363→   363→      throw new Error(`صندوق غير موجود (رقم ${cashboxId}) — أعد اختيار الصندوق`);
   364→   364→    }
   365→   365→    if (Number(box[0]!.is_archived) === 1) {
   366→   366→      throw new Error(`الصندوق «${box[0]!.name}» مؤرشف — اختر صندوقاً فاعلاً`);
   367→   367→    }
   368→   368→  }
   369→   369→
   370→   370→  // 8) فحص المخزون قبل أي كتابة (قرار 9 — لا مسار يحفظ وينقص سالباً)
   371→   371→  //    المسودة لا تُنقص شيئاً فلا تُفحص (الفحص وقت التحويل — FR-02-18)
   372→   372→  if (!isDraft) {
   373→   373→    await assertStockAvailable(db, v.items, v.warehouseId, products);
   374→   374→  }
   375→   375→
   376→   376→  // 9) حد الائتمان (FR-03-05) — للمكتملة الآجلة/المختلطة
   377→   377→  if (!isDraft && (v.payType === 'credit' || v.payType === 'mixed') && customerId !== null) {
   378→   378→    await assertCreditLimit(db, customerId, v.currencyId, due, v.creditLimitConfirmed === true);
   379→   379→  }
   380→   380→
   381→   381→  return {

   385→   385→    warehouseId: v.warehouseId,

   392→   392→    totals,
   393→   393→    payType: v.payType,

   397→   397→    cashboxId,
   398→   398→  };
   399→   399→}

   401→   401→/** جمع الكميات لكل صنف غير خدمي ومقارنتها برصيد المستودع — أول نقص يرمي. */
   402→   402→async function assertStockAvailable(
   403→   403→  db: DbEngine,
   404→   404→  items: SaleItemInput[],
   405→   405→  warehouseId: number,
   406→   406→  products: Map<number, ProductLite>,
   407→   407→): Promise<void> {
   408→   408→  const requested = new Map<number, Decimal>();
   409→   409→  for (const it of items) {
   410→   410→    if (it.productId === null) continue;
   411→   411→    const p = products.get(it.productId);
   412→   412→    if (p === undefined || p.isService) continue;
   413→   413→    requested.set(it.productId, (requested.get(it.productId) ?? dec(0)).plus(dec(it.qty)));
   414→   414→  }
   415→   415→  for (const [productId, qty] of requested) {
   416→   416→    const rows = await db.all<{ qty: string | number }>(
   417→   417→      'SELECT qty FROM stock_level WHERE product_id = ? AND warehouse_id = ?',
   418→   418→      [productId, warehouseId],
   419→   419→    );
   420→   420→    const available = dec(rows[0]?.qty ?? 0);
   421→   421→    if (qty.greaterThan(available)) {
   422→   422→      const p = products.get(productId)!;
   423→   423→      throw new StockShortageError(p.name, money(qty), money(available));
   424→   424→    }
   425→   425→  }
   426→   426→}

   428→   428→/** حد ائتمان العميل: 0 يمنع الآجل كلياً، والتجاوز حسب parties.credit_limit_action. */
   429→   429→async function assertCreditLimit(
   430→   430→  db: DbEngine,
   431→   431→  customerId: number,
   432→   432→  currencyId: number,
   433→   433→  due: Decimal,
   434→   434→  confirmed: boolean,
   435→   435→): Promise<void> {
   436→   436→  const rows = await db.all<{ name: string; credit_limit: string | number | null }>(
   437→   437→    'SELECT name, credit_limit FROM customer WHERE id = ?',
   438→   438→    [customerId],
   439→   439→  );
   440→   440→  const cust = rows[0];
   441→   441→  if (cust === undefined || cust.credit_limit === null) return; // null = بلا حد
   442→   442→  const limit = dec(cust.credit_limit);
   443→   443→  if (limit.isZero()) {
   444→   444→    throw new Error(
   445→   445→      `البيع الآجل ممنوع للعميل «${cust.name}» — حد الائتمان مضبوط على صفر. ` +
   446→   446→        'اختر الدفع النقدي أو عدّل حد الائتمان من ملف العميل',
   447→   447→    );
   448→   448→  }
   449→   449→  const balances = await customerBalances(customerId);
   450→   450→  const current = dec(balances.find((b) => b.currencyId === currencyId)?.balance ?? 0);
   451→   451→  const projected = current.plus(due);
   452→   452→  if (!projected.greaterThan(limit)) return;
   453→   453→
   454→   454→  const action = await getSetting('parties.credit_limit_action');
   455→   455→  if (action === 'block') {
   456→   456→    throw new Error(
   457→   457→      `تجاوز حد الائتمان للعميل «${cust.name}» (الحد ${money(limit)} والرصيد بعد الفاتورة ${money(projected)}) — ` +
   458→   458→        'الإعداد يمنع الحفظ. خفّض المبلغ الآجل أو عدّل الحد من ملف العميل',
   459→   459→    );
   460→   460→  }
   461→   461→  if (!confirmed) {
   462→   462→    throw new CreditLimitConfirmationRequiredError(cust.name, money(limit), money(projected), customerId);
   463→   463→  }
   464→   464→}

   493→   493→/** حركات المخزون (sale سالبة) + تحديث stock_level — للأصناف غير الخدمية فقط. */
   494→   494→async function applyStockOut(db: DbEngine, ctx: SaleCtx, invoiceId: number, items: SaleItemInput[]): Promise<void> {
   495→   495→  for (const it of items) {
   496→   496→    if (it.productId === null) continue;
   497→   497→    const p = ctx.products.get(it.productId);
   498→   498→    if (p === undefined || p.isService) continue;
   499→   499→    const qty = dec(it.qty).neg();
   500→   500→    await db.run(
   501→   501→      'INSERT INTO stock_movement(product_id, warehouse_id, movement_type, qty, unit_cost, ref_type, ref_id, moved_at, notes, created_at, created_by) ' +
   502→   502→        "VALUES(?, ?, 'sale', ?, ?, 'invoice', ?, ?, NULL, ?, ?)",
   503→   503→      [p.id, ctx.warehouseId, money(qty), money(p.costPrice), invoiceId, ctx.now, ctx.now, getCurrentUserId() ?? null],
   504→   504→    );
   505→   505→    // ملاحظة محرك: UPSERT بقيمة سالبة يفشل على CHECK qty>=0 قبل معالجة التعارض —
   506→   506→    // صف الرصيد موجود حتماً بعد فحص التوفر (متاح ≥ مطلوب > 0)، فيكفي UPDATE مباشر.
   507→   507→    const res = await db.run(
   508→   508→      'UPDATE stock_level SET qty = qty + ? WHERE product_id = ? AND warehouse_id = ?',
   509→   509→      [money(qty), p.id, ctx.warehouseId],
   510→   510→    );
   511→   511→    if (res.changes === 0) {
   512→   512→      // دفاعي (لا يُفترض الوصول إليه بعد فحص النقص) — قرار 9: لا مسار يحفظ سالباً
   513→   513→      throw new StockShortageError(p.name, money(dec(it.qty)), '0');
   514→   514→    }
   515→   515→  }
   516→   516→}
   517→   517→
   518→   518→/**
   519→   519→ * حركة القبض + التخصيص للجزء النقدي.
   520→   520→ * ملاحظة (قرار موثق أعلاه): customer_id=NULL — الجزء النقدي ليس تحصيلاً لدين سابق،
   521→   521→ * ومعادلة رصيد العميل تطرح القبضات المربوطة بالعميل فقط.
   522→   522→ */
   523→   523→async function applyCashIn(db: DbEngine, ctx: SaleCtx, invoiceId: number): Promise<void> {
   524→   524→  if (ctx.paid.lessThanOrEqualTo(0) || ctx.cashboxId === null) return;
   525→   525→  const res = await db.run(
   526→   526→    'INSERT INTO cash_tx(tx_type, cashbox_id, currency_id, amount, exchange_rate, tx_date, ref_type, ref_id, ' +
   527→   527→      'customer_id, is_voided, description, created_at, created_by) ' +
   528→   528→      "VALUES('receipt', ?, ?, ?, ?, ?, 'invoice', ?, NULL, 0, ?, ?, ?)",
   529→   529→    [
   530→   530→      ctx.cashboxId,
   531→   531→      ctx.currencyId,
   532→   532→      money(ctx.paid),
   533→   533→      ctx.rate,
   534→   534→      ctx.issuedAt,
   535→   535→      invoiceId,
   536→   536→      `تحصيل نقدي لفاتورة البيع رقم ${invoiceId}`,
   537→   537→      ctx.now,
   538→   538→      getCurrentUserId() ?? null,
   539→   539→    ],
   540→   540→  );
   541→   541→  await db.run(
   542→   542→    'INSERT INTO payment_allocation(cash_tx_id, invoice_id, allocated_amount, allocated_at, created_by) VALUES(?, ?, ?, ?, ?)',
   543→   543→    [Number(res.lastInsertRowId), invoiceId, money(ctx.paid), ctx.now, getCurrentUserId() ?? null],
   544→   544→  );
   545→   545→}
