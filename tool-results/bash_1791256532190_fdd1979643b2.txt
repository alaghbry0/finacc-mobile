    43→  readonly requested: string;
    44→  readonly available: string;

    46→    const missing = dec(requested).minus(dec(available));
    47→    super(
    48→      `كمية «${productName}» المطلوبة ${requested} والمتاح منها ${available} فقط — ينقص ${money(missing)}. ` +
    49→        'خفّض الكمية أو احذف الزيادة',
    50→    );
    51→    this.name = 'StockShortageError';
    52→    this.productName = productName;
    53→    this.requested = requested;
    54→    this.available = available;
    55→  }
    56→}

   175→  const parsed = SaveSaleSchema.safeParse(input);

   180→    );
   181→  }
   182→  return parsed.data;
   183→}

   204→interface SaleCtx {

   207→  issuedAt: string;

   209→  currencyId: number;
   210→  rate: string;
   211→  rateIsFallback: boolean;
   212→  taxMode: 'per_item' | 'on_total';
   213→  taxRate: string;
   214→  products: Map<number, ProductLite>;
   215→  totals: InvoiceTotals;
   216→  payType: SalePayType;

   223→async function loadProducts(db: DbEngine, items: SaleItemInput[]): Promise<Map<number, ProductLite>> {
   224→  const ids = [...new Set(items.filter((it) => it.productId !== null).map((it) => it.productId as number))];
   225→  const products = new Map<number, ProductLite>();
   226→  for (const id of ids) {
   227→    const rows = await db.all<{ id: number; name: string; cost_price: string | number | null; is_service: number; is_archived: number }>(
   228→      'SELECT id, name, cost_price, is_service, is_archived FROM product WHERE id = ?',
   229→      [id],
   230→    );
   231→    const r = rows[0];
   232→    if (r === undefined) {
   233→      throw new Error(`صنف غير موجود (رقم ${id}) — ربما حُذف رابطه، أعد إضافة الصنف للفاتورة`);
   234→    }
   235→    if (Number(r.is_archived) === 1) {
   236→      throw new Error(`الصنف «${r.name}» مؤرشف ولا يمكن بيعه — أعد تفعيله من شاشة المخزون أولاً`);
   237→    }
   238→    products.set(Number(r.id), {
   239→      id: Number(r.id),
   240→      name: r.name,
   241→      costPrice: dec(r.cost_price ?? 0),
   242→      isService: Number(r.is_service) === 1,
   243→    });
   244→  }
   245→  return products;
   246→}

   248→/** بناء السياق: تحقق + حسابات + سعر صرف + فترات + دفع + حد ائتمان (بلا كتابة). */
   249→async function buildSaleCtx(db: DbEngine, v: SaveSaleParsed, opts: { fromDraft?: boolean }): Promise<SaleCtx> {
   250→  const now = new Date().toISOString();
   251→  const issuedAt = v.issuedAt ?? todayISO();
   252→  const isDraft = v.saveAsDraft === true;
   253→
   254→  // 1) المراجع: مستودع/عملة (+ صندوق/عميل عند الحاجة)
   255→  const wh = await db.all<{ id: number; name: string }>(
   256→    'SELECT id, name FROM warehouse WHERE id = ? AND is_archived = 0',
   257→    [v.warehouseId],
   258→  );
   259→  if (wh.length === 0) {
   260→    throw new Error(`المستودع (رقم ${v.warehouseId}) غير موجود أو مؤرشف — اختر مستودعاً صالحاً`);
   261→  }
   262→  const cur = await db.all<{ id: number; code: string; is_active: number }>(
   263→    'SELECT id, code, is_active FROM currency WHERE id = ?',
   264→    [v.currencyId],
   265→  );
   266→  if (cur.length === 0 || Number(cur[0]!.is_active) !== 1) {
   267→    throw new Error(`العملة (رقم ${v.currencyId}) غير موجودة أو موقوفة — اختر عملة مفعلة`);
   268→  }
   269→
   270→  // 2) الأصناف والخدمات
   271→  const products = await loadProducts(db, v.items);
   272→  for (const it of v.items) {
   273→    if (it.productId === null && (it.lineDesc ?? '').trim().length === 0) {
   274→      throw new Error('سطر الخدمة الحرة يتطلب وصفاً — أدخل وصف البند (مثل «توصيل») أو اربطه بصنف');
   275→    }
   276→  }
   277→
   278→  // 3) الحسابات (computeInvoiceTotals فقط — لا حساب يدوي)
   279→  const taxModeRaw = await getSetting('invoicing.tax_mode');
   280→  const taxMode: 'per_item' | 'on_total' = taxModeRaw === 'per_item' ? 'per_item' : 'on_total';
   281→  const companyRows = await db.all<{ tax_rate: string | number | null }>('SELECT tax_rate FROM company LIMIT 1');
   282→  const taxRate = money(companyRows[0]?.tax_rate ?? 0);
   283→
   284→  const lines: InvoiceLineInput[] = v.items.map((it) => {
   285→    const p = it.productId !== null ? products.get(it.productId) : undefined;
   286→    const lineCost = p !== undefined && !p.isService ? money(roundTo(p.costPrice.times(dec(it.qty)), 4)) : '0';
   287→    return {
   288→      productId: it.productId,
   289→      lineDesc: it.lineDesc,
   290→      qty: it.qty,
   291→      unitPrice: it.unitPrice,
   292→      discountPercent: it.discountPercent ?? '0',
   293→      taxPercent: it.taxPercent ?? '0',
   294→      lineCost,
   295→    };
   296→  });
   297→  const totals = computeInvoiceTotals(lines, { taxMode, taxRate, invoiceDiscount: v.invoiceDiscount ?? '0' });
   298→  assertTotalsValid(totals); // منع صافي ≤ 0 (FR-02-05)
   299→
   300→  // 4) سعر الصرف (قرار 3): MissingRateError تمر كما هي — قبل أي كتابة
   301→  const base = await getBaseCurrency();
   302→  const snapshot = v.currencyId === base.id ? { rate: '1', rateIsFallback: false } : await getRateSnapshot(v.currencyId, issuedAt);
   303→
   304→  // 5) الفترات والتأريخ (قاعدة 5.4-11 + قرار 6) — assertBackdateAllowed يقيّد backdate في audit
   305→  await assertPeriodOpen(issuedAt);
   306→  await assertBackdateAllowed(issuedAt, { managerConfirmed: v.managerConfirmedBackdate });
   307→
   308→  const total = dec(totals.total);
   309→
   310→  // 6) تقسيم الدفع (المسودة: نوايا بلا متطلبات)
   311→  let paid = dec(0);
   312→  let due = dec(0);
   313→  if (isDraft) {
   314→    due = total;
   315→  } else if (v.payType === 'cash') {
   316→    paid = total;
   317→  } else if (v.payType === 'credit') {
   318→    due = total;
   319→  } else {
   320→    if (v.cashPart === undefined || v.cashPart === '') {
   321→      throw new Error('الفاتورة المختلطة تتطلب مبلغ الجزء النقدي — أدخل المبلغ المدفوع نقداً أو اختر نوع دفع آخر');
   322→    }
   323→    const cp = dec(v.cashPart);
   324→    if (cp.lessThanOrEqualTo(0)) {
   325→      throw new Error('الجزء النقدي في الفاتورة المختلطة يجب أن يكون أكبر من صفر — للآجل الكامل استخدم «حفظ آجل»');
   326→    }
   327→    if (cp.greaterThan(total)) {
   328→      throw new Error('الجزء النقدي أكبر من إجمالي الفاتورة — خفّض المبلغ أو استخدم الدفع النقدي الكامل');
   329→    }
   330→    paid = cp;
   331→    due = total.minus(cp);
   332→  }
   333→
   334→  // 7) متطلبات الطرف/الصندوق (للمكتملة فقط)
   335→  let customerId: number | null = v.customerId ?? null;
   336→  let cashboxId: number | null = v.cashboxId ?? null;
   337→  if (!isDraft) {
   338→    if ((v.payType === 'credit' || v.payType === 'mixed') && customerId === null) {
   339→      throw new Error('حفظ الفاتورة الآجلة يتطلب اختيار عميل أولاً — اختر العميل من شريط الأعلى أو استخدم الدفع النقدي');
   340→    }
   341→    if ((v.payType === 'cash' || v.payType === 'mixed') && cashboxId === null) {
   342→      throw new Error('الدفع النقدي يتطلب اختيار صندوق — اختر الصندوق من شريط الأعلى قبل الحفظ');
   343→    }
   344→  }
   345→  if (customerId !== null) {
   346→    const cust = await db.all<{ id: number; name: string; is_archived: number }>(
   347→      'SELECT id, name, is_archived FROM customer WHERE id = ?',
   348→      [customerId],
   349→    );
   350→    if (cust.length === 0) {
   351→      throw new Error(`عميل غير موجود (رقم ${customerId}) — أعد اختيار العميل من القائمة`);
   352→    }
   353→    if (Number(cust[0]!.is_archived) === 1) {
   354→      throw new Error(`العميل «${cust[0]!.name}» مؤرشف — أعد تفعيله من شاشة الأطراف أولاً`);
   355→    }
   356→  }
   357→  if (cashboxId !== null) {
   358→    const box = await db.all<{ id: number; name: string; is_archived: number }>(
   359→      'SELECT id, name, is_archived FROM cashbox WHERE id = ?',
   360→      [cashboxId],
   361→    );
   362→    if (box.length === 0) {
   363→      throw new Error(`صندوق غير موجود (رقم ${cashboxId}) — أعد اختيار الصندوق`);
   364→    }
   365→    if (Number(box[0]!.is_archived) === 1) {
   366→      throw new Error(`الصندوق «${box[0]!.name}» مؤرشف — اختر صندوقاً فاعلاً`);
   367→    }
   368→  }
   369→
   370→  // 8) فحص المخزون قبل أي كتابة (قرار 9 — لا مسار يحفظ وينقص سالباً)
   371→  //    المسودة لا تُنقص شيئاً فلا تُفحص (الفحص وقت التحويل — FR-02-18)
   372→  if (!isDraft) {
   373→    await assertStockAvailable(db, v.items, v.warehouseId, products);
   374→  }
   375→
   376→  // 9) حد الائتمان (FR-03-05) — للمكتملة الآجلة/المختلطة
   377→  if (!isDraft && (v.payType === 'credit' || v.payType === 'mixed') && customerId !== null) {
   378→    await assertCreditLimit(db, customerId, v.currencyId, due, v.creditLimitConfirmed === true);
   379→  }
   380→
   381→  return {
   382→    db,
   383→    now,
   384→    issuedAt,
   385→    warehouseId: v.warehouseId,
   386→    currencyId: v.currencyId,
   387→    rate: snapshot.rate,
   388→    rateIsFallback: snapshot.rateIsFallback,
   389→    taxMode,
   390→    taxRate,
   391→    products,
   392→    totals,
   393→    payType: v.payType,
   394→    paid,
   395→    due,
   396→    customerId,
   397→    cashboxId,
   398→  };
   399→}

   401→/** جمع الكميات لكل صنف غير خدمي ومقارنتها برصيد المستودع — أول نقص يرمي. */
   402→async function assertStockAvailable(
   403→  db: DbEngine,
   404→  items: SaleItemInput[],
   405→  warehouseId: number,
   406→  products: Map<number, ProductLite>,
   407→): Promise<void> {
   408→  const requested = new Map<number, Decimal>();
   409→  for (const it of items) {
   410→    if (it.productId === null) continue;
   411→    const p = products.get(it.productId);
   412→    if (p === undefined || p.isService) continue;
   413→    requested.set(it.productId, (requested.get(it.productId) ?? dec(0)).plus(dec(it.qty)));
   414→  }
   415→  for (const [productId, qty] of requested) {
   416→    const rows = await db.all<{ qty: string | number }>(
   417→      'SELECT qty FROM stock_level WHERE product_id = ? AND warehouse_id = ?',
   418→      [productId, warehouseId],
   419→    );
   420→    const available = dec(rows[0]?.qty ?? 0);
   421→    if (qty.greaterThan(available)) {
   422→      const p = products.get(productId)!;
   423→      throw new StockShortageError(p.name, money(qty), money(available));
   424→    }
   425→  }
   426→}

   428→/** حد ائتمان العميل: 0 يمنع الآجل كلياً، والتجاوز حسب parties.credit_limit_action. */
   429→async function assertCreditLimit(
   430→  db: DbEngine,
   431→  customerId: number,
   432→  currencyId: number,
   433→  due: Decimal,
   434→  confirmed: boolean,
   435→): Promise<void> {
   436→  const rows = await db.all<{ name: string; credit_limit: string | number | null }>(
   437→    'SELECT name, credit_limit FROM customer WHERE id = ?',
   438→    [customerId],
   439→  );
   440→  const cust = rows[0];
   441→  if (cust === undefined || cust.credit_limit === null) return; // null = بلا حد
   442→  const limit = dec(cust.credit_limit);
   443→  if (limit.isZero()) {
   444→    throw new Error(
   445→      `البيع الآجل ممنوع للعميل «${cust.name}» — حد الائتمان مضبوط على صفر. ` +
   446→        'اختر الدفع النقدي أو عدّل حد الائتمان من ملف العميل',
   447→    );
   448→  }
   449→  const balances = await customerBalances(customerId);
   450→  const current = dec(balances.find((b) => b.currencyId === currencyId)?.balance ?? 0);
   451→  const projected = current.plus(due);
   452→  if (!projected.greaterThan(limit)) return;
   453→
   454→  const action = await getSetting('parties.credit_limit_action');
   455→  if (action === 'block') {
   456→    throw new Error(
   457→      `تجاوز حد الائتمان للعميل «${cust.name}» (الحد ${money(limit)} والرصيد بعد الفاتورة ${money(projected)}) — ` +
   458→        'الإعداد يمنع الحفظ. خفّض المبلغ الآجل أو عدّل الحد من ملف العميل',
   459→    );
   460→  }
   461→  if (!confirmed) {
   462→    throw new CreditLimitConfirmationRequiredError(cust.name, money(limit), money(projected), customerId);
   463→  }
   464→}

   493→/** حركات المخزون (sale سالبة) + تحديث stock_level — للأصناف غير الخدمية فقط. */
   494→async function applyStockOut(db: DbEngine, ctx: SaleCtx, invoiceId: number, items: SaleItemInput[]): Promise<void> {
   495→  for (const it of items) {
   496→    if (it.productId === null) continue;
   497→    const p = ctx.products.get(it.productId);
   498→    if (p === undefined || p.isService) continue;
   499→    const qty = dec(it.qty).neg();
   500→    await db.run(
   501→      'INSERT INTO stock_movement(product_id, warehouse_id, movement_type, qty, unit_cost, ref_type, ref_id, moved_at, notes, created_at, created_by) ' +
   502→        "VALUES(?, ?, 'sale', ?, ?, 'invoice', ?, ?, NULL, ?, ?)",
   503→      [p.id, ctx.warehouseId, money(qty), money(p.costPrice), invoiceId, ctx.now, ctx.now, getCurrentUserId() ?? null],
   504→    );
   505→    // ملاحظة محرك: UPSERT بقيمة سالبة يفشل على CHECK qty>=0 قبل معالجة التعارض —
   506→    // صف الرصيد موجود حتماً بعد فحص التوفر (متاح ≥ مطلوب > 0)، فيكفي UPDATE مباشر.
   507→    const res = await db.run(
   508→      'UPDATE stock_level SET qty = qty + ? WHERE product_id = ? AND warehouse_id = ?',
   509→      oney(qty), p.id, ctx.warehouseId],
   510→    );
   511→    if (res.changes === 0) {
   512→      // دفاعي (لا يُفترض الوصول إليه بعد فحص النقص) — قرار 9: لا مسار يحفظ سالباً
   513→      throw new StockShortageError(p.name, money(dec(it.qty)), '0');
   514→    }
   515→  }
   516→}
   517→
   518→/**
   519→ * حركة القبض + التخصيص للجزء النقدي.
   520→ * ملاحظة (قرار موثق أعلاه): customer_id=NULL — الجزء النقدي ليس تحصيلاً لدين سابق،
   521→ * ومعادلة رصيد العميل تطرح القبضات المربوطة بالعميل فقط.
   522→ */
   523→async function applyCashIn(db: DbEngine, ctx: SaleCtx, invoiceId: number): Promise<void> {
   524→  if (ctx.paid.lessThanOrEqualTo(0) || ctx.cashboxId === null) return;
   525→  const res = await db.run(
   526→    'INSERT INTO cash_tx(tx_type, cashbox_id, currency_id, amount, exchange_rate, tx_date, ref_type, ref_id, ' +
   527→      'customer_id, is_voided, description, created_at, created_by) ' +
   528→      "VALUES('receipt', ?, ?, ?, ?, ?, 'invoice', ?, NULL, 0, ?, ?, ?)",
   529→    [
   530→      ctx.cashboxId,
   531→      ctx.currencyId,
   532→      money(ctx.paid),
   533→      ctx.rate,
   534→      ctx.issuedAt,
   535→      invoiceId,
   536→      `تحصيل نقدي لفاتورة البيع رقم ${invoiceId}`,
   537→      ctx.now,
   538→      getCurrentUserId() ?? null,
   539→    ],
   540→  );
   541→  await db.run(
   542→    'INSERT INTO payment_allocation(cash_tx_id, invoice_id, allocated_amount, allocated_at, created_by) VALUES(?, ?, ?, ?, ?)',
   543→    [Number(res.lastInsertRowId), invoiceId, money(ctx.paid), ctx.now, getCurrentUserId() ?? null],
   544→  );
   545→}

   753→  if (opts.managerConfirmed !== true) {

   759→async function voidInvoiceCore(invoiceId: number, opts: { managerConfirmed: boolean; reason?: string }): Promise<void> {
   760→  const db = await getDb();
   761→  await db.transaction(async () => {
   762→    const rows = await db.all<{
   763→      id: number; invoice_no: string | null; doc_type: string; status: string; customer_id: number | null;
   764→    }>('SELECT id, invoice_no, doc_type, status, customer_id FROM invoice WHERE id = ?', [invoiceId]);
   765→    const inv = rows[0];
   766→    if (inv === undefined) {
   767→      throw new Error(`فاتورة غير موجودة (رقم ${invoiceId}) — لا يمكن إلغاؤها`);
   768→    }
   769→    if (inv.status === 'void') {
   770→      throw new Error('هذه الفاتورة ملغاة مسبقاً — لا يمكن إلغاؤها مرتين');
   771→    }
   772→
   773→    const now = new Date().toISOString();
   774→
   775→    // المسودة: تُلغى بلا أثر أصلاً (تحويل الحالة فقط — قرار 2)
   776→    if (inv.status === 'draft') {
   777→      await db.run('UPDATE invoice SET status = ?, updated_at = ? WHERE id = ?', ['void', now, invoiceId]);
   778→      await logAudit('void_invoice', {
   779→        entity: 'invoice',
   780→        entityId: invoiceId,
   781→        details: { invoiceNo: inv.invoice_no, reason: opts.reason ?? null, wasDraft: true },
   782→      });
   783→      return;
   784→    }
   785→
   786→    // مرتجعات بيع مكتملة مرتبطة → رفض (ألغِ المرتجعات أولاً)
   787→    if (inv.doc_type === 'sale') {
   788→      const returns = await db.all<{ c: number }>(
   789→        "SELECT count(*) AS c FROM invoice WHERE original_invoice_id = ? AND doc_type = 'sale_return' AND status = 'completed'",
   790→        [invoiceId],
   791→      );
   792→      if (Number(returns[0]?.c ?? 0) > 0) {
   793→        throw new Error(
   794→          'لا يمكن إلغاء فاتورة عليها مرتجعات مكتملة — ألغِ المرتجعات المرتبطة بها أولاً ثم أعد محاولة إلغاء الفاتورة',
   795→        );
   796→      }
   797→    }
   798→
   799→    // 1) مخزون معاكس: لكل حركة sale سالبة مرتبطة بالفاتورة حركة موجبة بنفس unit_cost وref
   800→    const moves = await db.all<{ id: number; product_id: number; warehouse_id: number; qty: string | number; unit_cost: string | number }>(
   801→      "SELECT id, product_id, warehouse_id, qty, unit_cost FROM stock_movement WHERE ref_type = 'invoice' AND ref_id = ? AND movement_type = 'sale'",
   802→      [invoiceId],
   803→    );
   804→    for (const m of moves) {
   805→      const back = dec(m.qty).neg(); // موجبة
   806→      await db.run(
   807→        'INSERT INTO stock_movement(product_id, warehouse_id, movement_type, qty, unit_cost, ref_type, ref_id, moved_at, notes, created_at, created_by) ' +
   808→          "VALUES(?, ?, 'sale', ?, ?, 'invoice', ?, ?, ?, ?, ?)",
   809→        [
   810→          Number(m.product_id),
   811→          Number(m.warehouse_id),
   812→          money(back),
   813→          money(m.unit_cost),
   814→          invoiceId,
   815→          now,
   816→          `حركة معاكسة لإلغاء الفاتورة ${inv.invoice_no ?? invoiceId}`,
   817→          now,
   818→          getCurrentUserId() ?? null,
   819→        ],
   820→      );
   821→      const res = await db.run(
   822→        'UPDATE stock_level SET qty = qty + ? WHERE product_id = ? AND warehouse_id = ?',
   823→        oney(back), Number(m.product_id), Number(m.warehouse_id)],
   824→      );
   825→      if (res.changes === 0) {
   826→        // صف رصيد جديد (حركة موجبة — CHECK يسمح)
   827→        await db.run(
   828→          'INSERT INTO stock_level(product_id, warehouse_id, qty) VALUES(?, ?, ?)',
   829→          [Number(m.product_id), Number(m.warehouse_id), money(back)],
   830→        );
   831→      }
   832→    }
   833→

   838→    }>("SELECT * FROM cash_tx WHERE ref_type = 'invoice' AND ref_id = ? AND is_voided = 0", [invoiceId]);
   839→    for (const t of txs) {
   840→      await db.run('UPDATE cash_tx SET is_voided = 1 WHERE id = ?', [Number(t.id)]);
   841→      const oppositeType = t.tx_type === 'receipt' ? 'payment' : 'receipt';
   842→      await db.run(
   843→        'INSERT INTO cash_tx(tx_type, cashbox_id, currency_id, amount, exchange_rate, tx_date, ref_type, ref_id, ' +

   845→          'VALUES(?, ?, ?, ?, ?, ?, ?, ?, ?, ?, 0, ?, ?, ?, ?)',
   846→        [
   847→          oppositeType,
   848→          Number(t.cashbox_id),
   849→          Number(t.currency_id),
   850→          money(t.amount),
   851→          money(t.exchange_rate),
   852→          todayISO(),
   853→          'invoice',
   854→          invoiceId,
   855→          t.customer_id,
   856→          t.supplier_id,
   857→          Number(t.id),
   858→          `عكس حركة إلغاء الفاتورة ${inv.invoice_no ?? invoiceId}`,
   859→          now,
   860→          getCurrentUserId() ?? null,
   861→        ],
   862→      );
   863→    }
   864→
   865→    // 3) الحالة (يبقى الرقم ولا يُعاد أبداً — قرار 6)
   866→    await db.run('UPDATE invoice SET status = ?, updated_at = ? WHERE id = ?', ['void', now, invoiceId]);
   867→    await logAudit('void_invoice', {
   868→      entity: 'invoice',
   869→      entityId: invoiceId,
   870→      details: { invoiceNo: inv.invoice_no, reason: opts.reason ?? null },
   871→    });
   872→  });
   873→}
