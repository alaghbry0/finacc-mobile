    13→/**
    14→ * وحدة الشراء (FR-02-08 + FR-02-15 + قاعدة WAC 5.4-3) — Task 4-b:
    15→ *
    16→ * الشراء = مرآة البيع باتجاه معاكس داخل transaction واحدة:
    17→ *  - حركة مخزون 'purchase' كمية موجبة + زيادة stock_level.
    18→ *  - خصم رأس الفاتورة يوزَّع pro-rata على البنود قبل تحديث WAC (قاعدة 5.4-3):
    19→ *      distributeProRata(invoiceDiscount, lineNets) → unit_cost_net = (lineNet − allocated)/qty
    20→ *  - WAC الجديد: new_cost = (qty_old×cost_old + qty_new×unit_cost_net)/(qty_old+qty_new)
    21→ *      (qty_old ≤ 0 → unit_cost_net مباشرة) — qty_old هو الرصيد الكلي للصنف عبر كل
    22→ *      المخازن (cost_price حقل واحد على مستوى الصنف — قرار موثق).
    23→ *      كل الحسابات Decimal صريحة بتقريب 4 منازل (NUMERIC(14,4)).
    24→ *  - الدفع: cash/mixed → cash_tx tx_type='payment' (خروج من الصندوق) + payment_allocation؛
    25→ *      credit → due على المورد (رصيد المورد يزيد تلقائياً بمعادلة parties القرار 8 لأن
    26→ *      invoice.due_amount بعملة الفاتورة).
    27→ *      **قرار موثق**: cash_tx الجزء النقدي الفوري يُسجَّل بـ supplier_id=NULL — لقد دُفع
    28→ *      المبلغ لحظة الشراء ولم يُقيَّد AP أصلاً، وإسناده للمورد سيُخصم مرتين في معادلة
    29→ *      رصيد المورد (افتتاحي + Σ purchase.due − Σ payments − Σ purchase_return.due).
    30→ *      payment_allocation يحفظ الربط بالفاتورة للتوثيق.
    31→ *  - المسودة (draft): بلا رقم وبلا أثر مالي/مخزوني (قرار 2).
    32→ *
    33→ * voidPurchaseInvoice (FR-02-15): عكس كامل بلا حذف فيزيائي:
    34→ *  - حركات شراء معاكسة سالبة بنفس unit_cost الأصلية + إنقاص stock_level.
    35→ *  - **القرار الموثق لإرجاع WAC**: يُعاد حسابه رياضياً على الصافي المتبقي:
    36→ *        new_cost = (qty_total×wac_current − qty_inv×unit_cost_net)/(qty_total − qty_inv)
    37→ *    وهو يعيد تماماً قيمة ما قبل الشراء إن لم تدخل حركات أخرى، ويبقى صحيحاً رياضياً
    38→ *    بعد مبيعات واجهت WAC ثابتاً. إن كان المتبقي صفراً (رُجعت كل الكمية) تُبقى
    39→ *    التكلفة الحالية (لا أساس كمية لإعادة الحساب — ستُستبدل عند أول شراء قادم لأن
    40→ *    qty_old=0 يعتمد التكلفة الجديدة مباشرة). التفاصيل تُقيَّد في audit.
    41→ *  - عكس النقدية: UPDATE cash_tx SET is_voided=1 لكل سندات الصرف المرتبطة بالفاتورة
    42→ *    (**قرار موثق**: إبطال الصف نفسه بدل صف عكسي جديد — لا أثر مزدوج في رصيد الصندوق
    43→ *    ولا في معادلات الأطراف).
    44→ *  - يُرفض لو على الفاتورة مرتجعات شراء مكتملة (تُلغى المرتجعات أولاً — AC-21)،
    45→ *    أو لو كان المخزون الحالي أقل من كمية الفاتورة (سيخالف CHECK qty ≥ 0).
    46→ *  - رصيد المورد يرجع تلقائياً (المعادلة تعُدّ status='completed' فقط).
    47→ */

   206→/** قيمة اختيارية مُطهّرة: '' → undefined. */
   207→function normOpt(v: string | undefined | null): string | undefined {
   208→  if (v === undefined || v === null) return undefined;
   209→  const t = String(v).trim();
   210→  return t === '' ? undefined : t;

   213→/** إجمالي الكمية المتاحة لصنف عبر كل المخازن (أساس WAC — cost_price على مستوى الصنف). */
   214→async function totalStockOf(db: Awaited<ReturnType<typeof getDb>>, productId: number): Promise<Decimal> {
   215→  const rows = await db.all<{ q: string | number }>(
   216→    'SELECT COALESCE(SUM(qty), 0) AS q FROM stock_level WHERE product_id = ?',
   217→    [productId],
   218→  );
   219→  return dec(rows[0]?.q ?? 0);
   220→}

   222→/** تحديث رصيد مستودع لصنف (زيادة/إنقاص) — يُنادى داخل معاملة الحفظ. */
   223→async function bumpStockLevel(
   224→  db: Awaited<ReturnType<typeof getDb>>,
   225→  productId: number,
   226→  warehouseId: number,
   227→  delta: Decimal,
   228→): Promise<void> {
   229→  const rows = await db.all<{ qty: string | number }>(
   230→    'SELECT qty FROM stock_level WHERE product_id = ? AND warehouse_id = ?',
   231→    [productId, warehouseId],
   232→  );
   233→  const next = (rows.length > 0 ? dec(rows[0]!.qty) : dec(0)).plus(delta);
   234→  if (next.isNegative()) {
   235→    throw new Error('خطأ داخلي: محاولة إنقاص المخزون تحت الصفر — راجع كمية البنود');
   236→  }
   237→  await db.run(
   238→    'INSERT INTO stock_level(product_id, warehouse_id, qty) VALUES(?, ?, ?) ' +
   239→      'ON CONFLICT(product_id, warehouse_id) DO UPDATE SET qty = excluded.qty',
   240→    [productId, warehouseId, money(next)],
   241→  );
   242→}

   246→export async function savePurchaseInvoice(input: SavePurchaseInput): Promise<SavePurchaseResult> {
   247→  const v = SavePurchaseSchema.parse({
   248→    ...input,
   249→    supplierId: input.supplierId ?? null,
   250→    cashboxId: input.cashboxId ?? null,
   251→    cashPart: normOpt(input.cashPart),
   252→    invoiceDiscount: normOpt(input.invoiceDiscount),
   253→    notesInternal: normOpt(input.notesInternal),
   254→    notesPrinted: normOpt(input.notesPrinted),
   255→  });
   256→
   257→  const isDraft = v.saveAsDraft === true;
   258→
   259→  // متطلبات نوع الدفع (تتخفف للمسودة — لا أثر لها أصلاً)
   260→  if (!isDraft) {
   261→    if ((v.payType === 'credit' || v.payType === 'mixed') && (v.supplierId === null || v.supplierId === undefined)) {
   262→      throw new Error('فاتورة الشراء الآجلة/المختلطة تتطلب اختيار مورّد — اختر المورّد من القائمة أو حوّلها نقدياً');
   263→    }
   264→    if ((v.payType === 'cash' || v.payType === 'mixed') && (v.cashboxId === null || v.cashboxId === undefined)) {
   265→      throw new Error('الدفع النقدي يتطلب اختيار صندوق — اختر الصندوق الذي سيُصرف منه المبلغ');
   266→    }
   267→    if (v.payType === 'mixed' && v.cashPart === undefined) {
   268→      throw new Error('الفاتورة المختلطة تتطلب تحديد مبلغ الجزء النقدي — أدخل المبلغ المدفوع نقداً الآن');
   269→    }
   270→  }
   271→
   272→  const issuedAt = v.issuedAt ?? todayISO();
   273→  const db = await getDb();
   274→  const now = nowISO();
   275→  const userId = getCurrentUserId() ?? null;
   276→
   277→  let result: SavePurchaseResult = { invoiceId: 0, invoiceNo: null, total: '0', dueAmount: '0' };
   278→
   279→  await db.transaction(async () => {
   280→    // حجابا الفترات والتأريخ الرجعي (قاعدة 5.4-11) — قبل أي كتابة
   281→    await assertPeriodOpen(issuedAt);
   282→    await assertBackdateAllowed(issuedAt, { managerConfirmed: v.managerConfirmedBackdate === true });
   283→
   284→    // مراجع
   285→    const wh = await db.all<{ id: number }>('SELECT id FROM warehouse WHERE id = ?', [v.warehouseId]);
   286→    if (wh.length === 0) {
   287→      throw new Error(`المستودع (رقم ${v.warehouseId}) غير موجود — أعد اختيار المستودع`);
   288→    }
   289→    const cur = await db.all<{ id: number }>('SELECT id FROM currency WHERE id = ?', [v.currencyId]);
   290→    if (cur.length === 0) {
   291→      throw new Error(`العملة (رقم ${v.currencyId}) غير موجودة — أعد اختيار العملة`);
   292→    }
   293→    if (v.supplierId !== null && v.supplierId !== undefined) {
   294→      const sup = await db.all<{ id: number }>('SELECT id FROM supplier WHERE id = ?', [v.supplierId]);
   295→      if (sup.length === 0) {
   296→        throw new Error(`المورّد (رقم ${v.supplierId}) غير موجود — أعد اختيار المورّد`);
   297→      }
   298→    }
   299→    if (v.cashboxId !== null && v.cashboxId !== undefined) {
   300→      const box = await db.all<{ id: number; currency_id: number }>(
   301→        'SELECT id, currency_id FROM cashbox WHERE id = ?',
   302→        [v.cashboxId],
   303→      );
   304→      if (box.length === 0) {
   305→        throw new Error(`الصندوق (رقم ${v.cashboxId}) غير موجود — أعد اختيار الصندوق`);
   306→      }
   307→      if (Number(box[0]!.currency_id) !== v.currencyId) {
   308→        throw new Error('عملة الصندوق تختلف عن عملة الفاتورة — اختر صندوقاً بعملة الفاتورة (قرار 8: لا تُجمع عملات مختلفة)');
   309→      }
   310→    }
   311→
   312→    // الضريبة: نمط الإعدادات + نسبة الشركة
   313→    const taxMode = (await getSetting('invoicing.tax_mode')) === 'per_item' ? 'per_item' : 'on_total';
   314→    const companyRows = await db.all<{ tax_rate: string | number }>('SELECT tax_rate FROM company LIMIT 1');
   315→    const taxRate = money(companyRows[0]?.tax_rate ?? '0');
   316→
   317→    // الإجماليات عبر invoice-math (قاعدة إلزامية)
   318→    const totals = computeInvoiceTotals(
   319→      v.items.map((it) => ({
   320→        productId: it.productId,
   321→        lineDesc: it.lineDesc,
   322→        qty: it.qty,
   323→        unitPrice: it.unitPrice,
   324→        discountPercent: it.discountPercent,
   325→        taxPercent: it.taxPercent,
   326→      })),
   327→      { taxMode, taxRate, invoiceDiscount: v.invoiceDiscount ?? '0' },
   328→    );
   329→    assertTotalsValid(totals);
   330→
   331→    // توزيع خصم رأس الفاتورة pro-rata قبل تحديث WAC (قاعدة 5.4-3)
   332→    const allocs = distributeProRata(
   333→      v.invoiceDiscount ?? '0',
   334→      totals.lines.map((l) => l.lineNet),
   335→    );
   336→
   337→    const total = dec(totals.total);
   338→
   339→    // المدفوع/المتبقي حسب نوع الدفع
   340→    let paid = dec(0);
   341→    if (!isDraft) {
   342→      if (v.payType === 'cash') {
   343→        paid = total;
   344→      } else if (v.payType === 'mixed') {
   345→        paid = dec(v.cashPart!);
   346→        if (paid.greaterThan(total)) {
   347→          throw new Error(
   348→            `الجزء النقدي (${paid.toFixed(2)}) أكبر من إجمالي الفاتورة (${total.toFixed(2)}) — صحّح المبلغ`,
   349→          );
   350→        }
   351→      }
   352→    }
   353→    const due = isDraft ? dec(0) : total.minus(paid); // المسودة بلا أثر — لا دين ولا مدفوع
   354→
   355→    // لقطة سعر الصرف (قرار 3) — حتى للمسودة (عمود NOT NULL)
   356→    const rate = await getRateSnapshot(v.currencyId, issuedAt);
   357→    const totalBase = roundTo(total.times(dec(rate.rate)), 4);
   358→
   359→    // الرقم يُستهلك للمكتملة فقط (قرار 2)
   360→    const invoiceNo = isDraft ? null : (await nextDocNumber('PUR', { date: issuedAt })).docNo;
   361→
   362→    const invRes = await db.run(
   363→      'INSERT INTO invoice(invoice_no, doc_type, pay_status, status, issued_at, supplier_id, cashbox_id, warehouse_id, ' +
   364→        'currency_id, exchange_rate, rate_is_fallback, subtotal, discount_amount, tax_rate, tax_amount, total, total_base, ' +
   365→        'paid_amount, due_amount, cost_total, notes_internal, notes_printed, created_at, updated_at, created_by) ' +
   366→        "VALUES(?, 'purchase', ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)",
   367→      [
   368→        invoiceNo,
   369→        v.payType,
   370→        isDraft ? 'draft' : 'completed',
   371→        issuedAt,
   372→        v.supplierId ?? null,
   373→        !isDraft && (v.payType === 'cash' || v.payType === 'mixed') ? (v.cashboxId ?? null) : null,
   374→        v.warehouseId,
   375→        v.currencyId,
   376→        rate.rate,
   377→        rate.rateIsFallback ? 1 : 0,
   378→        totals.subtotal,
   379→        totals.invoiceDiscount,
   380→        taxRate,
   381→        totals.taxAmount,
   382→        totals.total,
   383→        money(totalBase),
   384→        money(roundTo(paid, 4)),
   385→        money(roundTo(due, 4)),
   386→        totals.costTotal,
   387→        v.notesInternal ?? null,
   388→        v.notesPrinted ?? null,
   389→        now,
   390→        now,
   391→        userId,
   392→      ],
   393→    );
   394→    const invoiceId = Number(invRes.lastInsertRowId);
   395→
   396→    // البنود + التكلفة الصافية (بعد التوزيع) + WAC + الحركات المخزونية
   397→    let costTotal = dec(0);
   398→    for (let i = 0; i < v.items.length; i++) {
   399→      const it = v.items[i]!;
   400→      const line = totals.lines[i]!;
   401→      const qty = dec(it.qty);
   402→      const allocated = dec(allocs[i]!);
   403→      const netCost = roundTo(dec(line.lineNet).minus(allocated), 4); // التكلفة الصافية للسطر
   404→      const unitCostNet = qty.greaterThan(0) ? netCost.div(qty) : dec(0);
   405→
   406→      let unitId: number | null = null;
   407→      if (it.productId !== null) {
   408→        const prod = await db.all<{
   409→          id: number;
   410→          name: string;
   411→          cost_price: string | number;
   412→          is_service: number;
   413→          unit_id: number | null;
   414→        }>('SELECT id, name, cost_price, is_service, unit_id FROM product WHERE id = ?', [it.productId]);
   415→        if (prod.length === 0) {
   416→          throw new Error(`الصنف (رقم ${it.productId}) غير موجود في بند رقم ${i + 1} — أعد اختيار الصنف`);
   417→        }
   418→        unitId = prod[0]!.unit_id === null ? null : Number(prod[0]!.unit_id);
   419→
   420→        if (!isDraft) {
   421→          const isService = Number(prod[0]!.is_service) === 1;
   422→          if (!isService) {
   423→            // WAC الجديد (قاعدة 5.4-3) — Decimal صريح بتقريب 4 منازل
   424→            const qtyOld = await totalStockOf(db, it.productId);
   425→            const costOld = dec(prod[0]!.cost_price);
   426→            let newCost: Decimal;
   427→            if (qtyOld.lte(0)) {
   428→              newCost = unitCostNet; // qty_old ≤ 0 → التكلفة الجديدة مباشرة
   429→            } else {
   430→              newCost = roundTo(qtyOld.times(costOld).plus(qty.times(unitCostNet)).div(qtyOld.plus(qty)), 4);
   431→            }
   432→            await db.run('UPDATE product SET cost_price = ?, updated_at = ? WHERE id = ?', [
   433→              money(newCost),
   434→              now,
   435→              it.productId,
   436→            ]);
   437→            await bumpStockLevel(db, it.productId, v.warehouseId, qty);
   438→            await db.run(
   439→              'INSERT INTO stock_movement(product_id, warehouse_id, movement_type, qty, unit_cost, ref_type, ref_id, moved_at, notes, created_at, created_by) ' +
   440→                "VALUES(?, ?, 'purchase', ?, ?, 'invoice', ?, ?, NULL, ?, ?)",
   441→              [it.productId, v.warehouseId, money(qty), money(roundTo(unitCostNet, 4)), invoiceId, issuedAt, now, userId],
   442→            );
   443→          }
   444→        }
   445→      }
   446→
   447→      await db.run(
   448→        'INSERT INTO invoice_item(invoice_id, product_id, line_desc, qty, unit_id, unit_factor, unit_price, ' +
   449→          'discount_percent, discount_amount, tax_percent, line_total, line_cost, notes, created_at) ' +
   450→          'VALUES(?, ?, ?, ?, ?, 1, ?, ?, ?, ?, ?, ?, NULL, ?)',
   451→        [
   452→          invoiceId,
   453→          it.productId,
   454→          it.lineDesc ?? null,
   455→          money(qty),
   456→          unitId,
   457→          it.unitPrice,
   458→          it.discountPercent ?? '0',
   459→          line.lineDiscount,
   460→          it.taxPercent ?? '0',
   461→          line.lineTotal,
   462→          money(netCost),
   463→          now,
   464→        ],
   465→      );
   466→      costTotal = costTotal.plus(netCost);
   467→    }
   468→    await db.run('UPDATE invoice SET cost_total = ? WHERE id = ?', [money(roundTo(costTotal, 4)), invoiceId]);
   469→
   470→    // النقدية: صرف من الصندوق (cash/mixed) + ربط التخصيص — supplier_id=NULL (قرار موثق أعلاه)
   471→    if (!isDraft && paid.greaterThan(0)) {
   472→      const txRes = await db.run(
   473→        'INSERT INTO cash_tx(tx_type, cashbox_id, currency_id, amount, exchange_rate, tx_date, ref_type, ref_id, ' +
   474→          'supplier_id, description, created_at, created_by) ' +
   475→        "VALUES('payment', ?, ?, ?, ?, ?, 'invoice', ?, NULL, ?, ?, ?)",
   476→        [
   477→          v.cashboxId!,
   478→          v.currencyId,
   479→          money(roundTo(paid, 4)),
   480→          rate.rate,
   481→          issuedAt,
   482→          invoiceId,
   483→          `دفع فاتورة شراء ${invoiceNo ?? ''}`.trim(),
   484→          now,
   485→          userId,
   486→        ],
   487→      );
   488→      await db.run(
   489→        'INSERT INTO payment_allocation(cash_tx_id, invoice_id, allocated_amount, allocated_at, created_by) VALUES(?, ?, ?, ?, ?)',
   490→        [Number(txRes.lastInsertRowId), invoiceId, money(roundTo(paid, 4)), issuedAt, userId],
   491→      );
   492→    }
   493→
   494→    await logAudit(isDraft ? 'purchase_draft_saved' : 'purchase_saved', {
   495→      entity: 'invoice',
   496→      entityId: invoiceId,
   497→      details: {
   498→        invoiceNo,
   499→        payType: v.payType,
   500→        total: totals.total,
   501→        paid: money(roundTo(paid, 4)),
   502→        due: money(roundTo(due, 4)),
   503→        supplierId: v.supplierId ?? null,
   504→        items: v.items.length,
   505→      },
   506→    });
   507→
   508→    result = {
   509→      invoiceId,
   510→      invoiceNo,
   511→      total: money(totals.total),
   512→      dueAmount: money(roundTo(due, 4)),
   513→    };
   514→  });
   515→
   516→  return result;
   517→}

   521→export async function voidPurchaseInvoice(
   522→  invoiceId: number,
   523→  opts: { managerConfirmed: boolean; reason?: string },
   524→): Promise<void> {
   525→  if (opts.managerConfirmed !== true) {
   526→    throw new Error('إلغاء فاتورة الشراء يتطلب تأكيد المدير — أكد الإلغاء من نافذة التأكيد أولاً');
   527→  }
   528→  const db = await getDb();
   529→  const now = nowISO();
   530→
   531→  await db.transaction(async () => {
   532→    const rows = await db.all<{
   533→      id: number;
   534→      doc_type: string;
   535→      invoice_no: string | null;
   536→      status: string;
   537→      issued_at: string;
   538→      supplier_id: number | null;
   539→      cashbox_id: number | null;
   540→      currency_id: number;
   541→    }>(
   542→      'SELECT id, doc_type, invoice_no, status, issued_at, supplier_id, cashbox_id, currency_id FROM invoice WHERE id = ?',
   543→      [invoiceId],
   544→    );
   545→    if (rows.length === 0) {
   546→      throw new Error(`فاتورة الشراء (رقم ${invoiceId}) غير موجودة`);
   547→    }
   548→    const inv = rows[0]!;
   549→    if (inv.doc_type !== 'purchase') {
   550→      throw new Error('هذا المستند ليس فاتورة شراء — الإلغاء يُنفَّذ من شاشته الخاصة بنوعه');
   551→    }
   552→    if (inv.status === 'void') {
   553→      throw new Error('الفاتورة ملغاة مسبقاً — لا يمكن تكرار الإلغاء');
   554→    }
   555→    if (inv.status === 'draft') {
   556→      throw new Error('لا يمكن إلغاء مسودة — للمسودة أثر مالي أو مخزوني أصلاً؛ أنشئ فاتورة جديدة عند الحاجة');
   557→    }
   558→
   559→    // رفض لو عليها مرتجعات شراء مكتملة (FR-02-15)
   560→    const linked = await db.all<{ c: number }>(
   561→      "SELECT count(*) AS c FROM invoice WHERE original_invoice_id = ? AND doc_type = 'purchase_return' AND status = 'completed'",
   562→      [invoiceId],
   563→    );
   564→    if (Number(linked[0]?.c ?? 0) > 0) {
   565→      throw new Error(
   566→        'لا يمكن إلغاء الفاتورة: عليها مرتجعات شراء مكتملة — راجع المرتجعات المرتبطة من شاشة التفاصيل أولاً',
   567→      );
   568→    }
   569→
   570→    // عكس المخزون: حركات معاكسة سالبة بنفس unit_cost الأصلية (Snapshot)
   571→    const movements = await db.all<{
   572→      id: number;
   573→      product_id: number;
   574→      warehouse_id: number;
   575→      qty: string | number;
   576→      unit_cost: string | number;
   577→    }>(
   578→      "SELECT id, product_id, warehouse_id, qty, unit_cost FROM stock_movement " +
   579→        "WHERE ref_type = 'invoice' AND ref_id = ? AND movement_type = 'purchase'",
   580→      [invoiceId],
   581→    );
   582→
   583→    // تجميع أثر كل صنف لإعادة حساب WAC دفعة واحدة (منتج قد يرد في عدة بنود)
   584→    const perProduct = new Map<
   585→      number,
   586→      { qtyInv: Decimal; valueInv: Decimal; warehouses: { warehouseId: number; qty: Decimal }[] }
   587→    >();
   588→    for (const m of movements) {
   589→      const pid = Number(m.product_id);
   590→      const qty = dec(m.qty);
   591→      const unit = dec(m.unit_cost);
   592→      const e = perProduct.get(pid) ?? { qtyInv: dec(0), valueInv: dec(0), warehouses: [] };
   593→      e.qtyInv = e.qtyInv.plus(qty);
   594→      e.valueInv = e.valueInv.plus(qty.times(unit));
   595→      const wid = Number(m.warehouse_id);
   596→      const existing = e.warehouses.find((w) => w.warehouseId === wid);
   597→      if (existing) existing.qty = existing.qty.plus(qty);
   598→      else e.warehouses.push({ warehouseId: wid, qty });
   599→      perProduct.set(pid, e);
   600→    }
   601→
   602→    const today = todayISO();
   603→    for (const [pid, agg] of perProduct) {
   604→      const prod = await db.all<{ name: string; cost_price: string | number }>(
   605→        'SELECT name, cost_price FROM product WHERE id = ?',
   606→        [pid],
   607→      );
   608→      const productName = prod[0]?.name ?? `#${pid}`;
   609→
   610→      for (const w of agg.warehouses) {
   611→        const lv = await db.all<{ qty: string | number }>(
   612→          'SELECT qty FROM stock_level WHERE product_id = ? AND warehouse_id = ?',
   613→          [pid, w.warehouseId],
   614→        );
   615→        const current = lv.length > 0 ? dec(lv[0]!.qty) : dec(0);
   616→        if (current.lessThan(w.qty)) {
   617→          throw new Error(
   618→            `لا يمكن الإلغاء: الصنف «${productName}» رصيده الحالي (${money(current)}) أقل من كمية الفاتورة (${money(w.qty)}) — ` +
   619→              'راجع حركات البيع أو أنشئ تسوية جرد أولاً',
   620→          );
   621→        }
   622→      }
   623→
   624→      // القرار الموثق لإرجاع WAC: إعادة حساب رياضية على الصافي المتبقي
   625→      const qtyTotal = await totalStockOf(db, pid);
   626→      const costNow = dec(prod[0]?.cost_price ?? 0);
   627→      const remaining = qtyTotal.minus(agg.qtyInv);
   628→      let newCost = costNow;
   629→      if (remaining.greaterThan(0)) {
   630→        newCost = roundTo(qtyTotal.times(costNow).minus(agg.valueInv).div(remaining), 4);
   631→      } // remaining = 0 → تُبقى التكلفة الحالية (لا أساس كمي — موثق في رأس الملف)
   632→
   633→      await db.run('UPDATE product SET cost_price = ?, updated_at = ? WHERE id = ?', [money(newCost), now, pid]);
   634→    }
   635→
   636→    // سطور العكس المخزوني (حركة شراء سالبة بنفس التكلفة — القرار الموثق)
   637→    for (const m of movements) {
   638→      const qty = dec(m.qty).neg();
   639→      await db.run(
   640→        'INSERT INTO stock_movement(product_id, warehouse_id, movement_type, qty, unit_cost, ref_type, ref_id, moved_at, notes, created_at, created_by) ' +
   641→          "VALUES(?, ?, 'purchase', ?, ?, 'invoice', ?, ?, ?, ?, ?)",
   642→        [
   643→          Number(m.product_id),
   644→          Number(m.warehouse_id),
   645→          money(qty),
   646→          money(dec(m.unit_cost)),
   647→          invoiceId,
   648→          today,
   649→          `عكس شراء بإلغاء الفاتورة ${inv.invoice_no ?? invoiceId}${opts.reason ? ` — ${opts.reason}` : ''}`,
   650→          now,
   651→          getCurrentUserId() ?? null,
   652→        ],
   653→      );
   654→      await bumpStockLevel(db, Number(m.product_id), Number(m.warehouse_id), qty);
   655→    }
   656→
   657→    // عكس النقدية: إبطال سندات الصرف المرتبطة (قرار موثق — لا صفوف عكسية جديدة)
   658→    await db.run(
   659→      "UPDATE cash_tx SET is_voided = 1 WHERE ref_type = 'invoice' AND ref_id = ? AND is_voided = 0",
   660→      [invoiceId],
   661→    );
   662→
   663→    // الحالة → void (الرقم لا يُعاد أبداً — 5.4-1) ورصيد المورد يرجع تلقائياً بالمعادلة
   664→    await db.run("UPDATE invoice SET status = 'void', updated_at = ? WHERE id = ?", [now, invoiceId]);
   665→
   666→    await logAudit('void_purchase_invoice', {
   667→      entity: 'invoice',
   668→      entityId: invoiceId,
   669→      details: {
   670→        invoiceNo: inv.invoice_no,
   671→        reason: opts.reason ?? null,
   672→        reversedMovements: movements.length,
   673→        wacRecalc: [...perProduct.entries()].map(([pid, agg]) => ({
   674→          productId: pid,
   675→          qtyInv: money(agg.qtyInv),
   676→          valueInv: money(roundTo(agg.valueInv, 4)),
   677→        })),
   678→      },
   679→    });
   680→  });
   681→}
