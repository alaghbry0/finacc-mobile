   287→   287→// ============ جمع دفتر صندوق (بعملته) ============
   288→   288→
   289→   289→/**
   290→   290→ * Σ الوارد والصادر لصندوق بعملته خلال نافذة اختيارية — كل الجمع Decimal في JS
   291→   291→ * (SUM في SQLite يُرجع REAL وقد يفقد الدقة). المستبعد دائماً: is_voided=1
   292→   292→ * أو reversal_of NOT NULL (اصطلاح الإلغاء أعلاه).
   293→   293→ */
   294→   294→async function sumBoxFlows(
   295→   295→  db: DbEngine,
   296→   296→  cashboxId: number,
   297→   297→  filter: { dateFrom?: string; dateTo?: string } = {},
   298→   298→): Promise<{ in: Decimal; out: Decimal }> {
   299→   299→  const boxRows = await db.all<{ currency_id: number }>('SELECT currency_id FROM cashbox WHERE id = ?', [cashboxId]);
   300→   300→  if (boxRows.length === 0) {
   301→   301→    throw new Error(`صندوق غير موجود (رقم ${cashboxId}) — لا يمكن حساب رصيده`);
   302→   302→  }
   303→   303→  const currencyId = Number(boxRows[0].currency_id);
   304→   304→
   305→   305→  const where: string[] = [
   306→   306→    't.is_voided = 0',
   307→   307→    't.reversal_of IS NULL',
   308→   308→    't.currency_id = ?',
   309→   309→    '(t.cashbox_id = ? OR (t.tx_type = \'box_transfer\' AND t.to_cashbox_id = ?))',
   310→   310→  ];
   311→   311→  const params: unknown[] = [currencyId, cashboxId, cashboxId];
   312→   312→  if (filter.dateFrom !== undefined) {
   313→   313→    where.push('t.tx_date >= ?');
   314→   314→    params.push(filter.dateFrom);
   315→   315→  }
   316→   316→  if (filter.dateTo !== undefined) {
   317→   317→    where.push('t.tx_date <= ?');
   318→   318→    params.push(filter.dateTo);
   319→   319→  }
   320→   320→
   321→   321→  const rows = await db.all<{ tx_type: string; cashbox_id: number; amount: string | number }>(
   322→   322→    `SELECT t.tx_type, t.cashbox_id, t.amount FROM cash_tx t WHERE ${where.join(' AND ')}`,
   323→   323→    params,
   324→   324→  );
   325→   325→
   326→   326→  let inSum = dec(0);
   327→   327→  let outSum = dec(0);
   328→   328→  for (const r of rows) {
   329→   329→    const amount = dec(r.amount);
   330→   330→    if (r.tx_type === 'box_transfer') {
   331→   331→      // ساق صادرة (cashbox_id=هذا) أو واردة (to_cashbox_id=هذا) — بعملة هذا الصندوق حصراً
   332→   332→      if (Number(r.cashbox_id) === cashboxId) outSum = outSum.plus(amount);
   333→   333→      else inSum = inSum.plus(amount);
   334→   334→    } else if (IN_TYPES.has(r.tx_type)) {
   335→   335→      inSum = inSum.plus(amount);
   336→   336→    } else if (OUT_TYPES.has(r.tx_type)) {
   337→   337→      outSum = outSum.plus(amount);
   338→   338→    }
   339→   339→  }
   340→   340→  return { in: inSum, out: outSum };
   341→   341→}

   345→   345→/** أرصدة كل الصناديق الفاعلة بعملاتها — السالب مسموح ويعود كما هو (قرار 9). */
   346→   346→export async function cashboxBalances(): Promise<CashboxBalanceRow[]> {
   347→   347→  const db = await getDb();
   348→   348→  const boxes = await db.all<{
   349→   349→    id: number; name: string; currency_id: number; is_default: number; code: string; decimals: number;
   350→   350→  }>(
   351→   351→    'SELECT cb.id, cb.name, cb.currency_id, cb.is_default, c.code, c.decimals ' +
   352→   352→      'FROM cashbox cb JOIN currency c ON c.id = cb.currency_id WHERE cb.is_archived = 0 ' +
   353→   353→      'ORDER BY cb.is_default DESC, cb.name COLLATE NOCASE ASC',
   354→   354→  );
   355→   355→  const out: CashboxBalanceRow[] = [];
   356→   356→  for (const b of boxes) {
   357→   357→    const flows = await sumBoxFlows(db, Number(b.id), {});
   358→   358→    out.push({
   359→   359→      id: Number(b.id),
   360→   360→      name: b.name,
   361→   361→      currencyId: Number(b.currency_id),
   362→   362→      currencyCode: b.code,
   363→   363→      currencyDecimals: Number(b.decimals),
   364→   364→      isDefault: Number(b.is_default) === 1,
   365→   365→      balance: money(flows.in.minus(flows.out)),
   366→   366→    });
   367→   367→  }
   368→   368→  return out;

   396→   396→// ============ المتبقي على فاتورة + FIFO (قاعدة 5.4-6) ============
   397→   397→
   398→   398→/**
   399→   399→ * المتبقي على فاتورة = due_amount − Σ تخصيصات سنداتها التي تحمل الطرف
   400→   400→ * (customer_id لفواتير البيع / supplier_id لفواتير الشراء) وغير الملغاة.
   401→   401→ * سند لحظة البيع النقدي لا يحمل الطرف (قرار invoicing) فلا يُخصم مرتين.
   402→   402→ */
   403→   403→async function invoiceRemainingDue(
   404→   404→  db: DbEngine,
   405→   405→  invoiceId: number,
   406→   406→  partyCol: 'customer_id' | 'supplier_id',
   407→   407→): Promise<Decimal> {
   408→   408→  const inv = (await db.all<{ due_amount: string | number }>('SELECT due_amount FROM invoice WHERE id = ?', [invoiceId]))[0];
   409→   409→  if (inv === undefined) {
   410→   410→    throw new Error(`فاتورة غير موجودة (رقم ${invoiceId}) — ربما حُذف الرابط، أعد اختيار الفاتورة`);
   411→   411→  }
   412→   412→  const allocs = await db.all<{ a: string | number }>(
   413→   413→    `SELECT pa.allocated_amount AS a FROM payment_allocation pa JOIN cash_tx t ON t.id = pa.cash_tx_id ` +
   414→   414→      `WHERE pa.invoice_id = ? AND t.${partyCol} IS NOT NULL AND t.is_voided = 0`,
   415→   415→    [invoiceId],
   416→   416→  );
   417→   417→  let allocated = dec(0);
   418→   418→  for (const r of allocs) allocated = allocated.plus(dec(r.a));
   419→   419→  return dec(inv.due_amount).minus(allocated);
   420→   420→}
   421→   421→
   422→   422→/**
   423→   423→ * تخصيص مبلغ قبض على الحساب FIFO: أقدم فواتير البيع المكتملة الآجلة ذات متبقٍ
   424→   424→ * للعميل ← payment_allocation حتى نفاد المبلغ. البقايا تبقى غير مخصصة = رصيد دائن.
   425→   425→ */
   426→   426→async function allocateReceiptFifo(
   427→   427→  db: DbEngine,
   428→   428→  cashTxId: number,
   429→   429→  customerId: number,
   430→   430→  amount: Decimal,
   431→   431→  now: string,
   432→   432→): Promise<void> {
   433→   433→  const existing = await db.all<{ a: string | number }>(
   434→   434→    'SELECT allocated_amount AS a FROM payment_allocation WHERE cash_tx_id = ?',
   435→   435→    [cashTxId],
   436→   436→  );
   437→   437→  let left = dec(amount);
   438→   438→  for (const r of existing) left = left.minus(dec(r.a));
   439→   439→  if (left.lessThanOrEqualTo(0)) return;
   440→   440→
   441→   441→  const invoices = await db.all<{ id: number; due_amount: string | number }>(
   442→   442→    "SELECT id, due_amount FROM invoice WHERE customer_id = ? AND doc_type = 'sale' AND status = 'completed' " +
   443→   443→      "AND pay_status IN ('credit','mixed') AND due_amount > 0 ORDER BY issued_at ASC, id ASC",
   444→   444→    [customerId],
   445→   445→  );
   446→   446→
   447→   447→  for (const inv of invoices) {
   448→   448→    if (left.lessThanOrEqualTo(0)) break;
   449→   449→    const remaining = (await invoiceRemainingDue(db, Number(inv.id), 'customer_id')).toDecimalPlaces(4);
   450→   450→    if (remaining.lessThanOrEqualTo(0)) continue;
   451→   451→    const alloc = Decimal.min(remaining, left.toDecimalPlaces(4));
   452→   452→    await db.run(
   453→   453→      'INSERT INTO payment_allocation(cash_tx_id, invoice_id, allocated_amount, allocated_at, created_by) VALUES(?, ?, ?, ?, ?)',
   454→   454→      [cashTxId, Number(inv.id), money(alloc), now, getCurrentUserId() ?? null],
   455→   455→    );
   456→   456→    left = left.minus(alloc);
   457→   457→  }
   458→   458→}

   694→   694→              `المبلغ يتجاوز المتبقي على فاتورة الشراء (المتبقي ${money(remaining)}) — خفّض المبلغ أو ادفع الباقي على الحساب`,
   695→   695→            );
   696→   696→          }
   697→   697→        }
   698→   698→      } else if (refType === 'on_account' || refType === 'invoice') {
   699→   699→        throw new Error('الدفع على الحساب أو على فاتورة محددة يتطلب اختيار المورّد أولاً');
   700→   700→      }
   701→   701→    }
   702→   702→
   703→   703→    if (v.txType === 'owner_draw' || v.txType === 'capital_in' || v.txType === 'bank_deposit' || v.txType === 'bank_withdraw' || v.txType === 'opening') {
   704→   704→      if (customerId !== null || supplierId !== null) {
   705→   705→        throw new Error('هذا النوع من الحركات لا يرتبط بعميل أو مورّد — أزل الطرف أو غيّر نوع الحركة');
   706→   706→      }
   707→   707→      customerId = null;
   708→   708→      supplierId = null;
   709→   709→      refType = null;
   710→   710→      refId = null;
   711→   711→    }
   712→   712→
   713→   713→    // 5) حسابات التحويل بين عملتين (قرار 8 / FR-04-07)
   714→   714→    let toAmount: Decimal | null = null;
   715→   715→    let rateDst: string | null = null;
   716→   716→    let fx = dec(0);
   717→   717→    if (v.txType === 'box_transfer' && destBox !== null) {
   718→   718→      refType = 'transfer';
   719→   719→      refId = null; // يُربط بالساق الأولى بعد إدراجها (مفتاح الزوج)
   720→   720→      if (customerId !== null || supplierId !== null) {
   721→   721→        throw new Error('التحويل بين صندوقين لا يرتبط بأطراف — أزل العميل/المورّد');
   722→   722→      }
   723→   723→      const sameCurrency = Number(destBox.currency_id) === v.currencyId;
   724→   724→      rateDst = sameCurrency
   725→   725→        ? rateSrc.rate
   726→   726→        : (await getRateSnapshot(Number(destBox.currency_id), txDate)).rate;
   727→   727→      const amountBase = amount.times(dec(rateSrc.rate)); // قيمة المبلغ بالعملة الأساسية
   728→   728→      toAmount =
   729→   729→        v.toAmount !== undefined
   730→   730→          ? dec(v.toAmount)
   731→   731→          : roundTo(amountBase.div(dec(rateDst)), Number(destBox.decimals));
   732→   732→      if (toAmount.lessThanOrEqualTo(0)) {
   733→   733→        throw new Error('المبلغ المحوّل بعملة الوجهة يجب أن يكون أكبر من صفر — راجع سعر الصرف أو المبلغ المدخل');
   734→   734→      }
   735→   735→      // فرق الصرف المحقق (بقيمة الأساس): قيمة ما وصل − قيمة ما خرج (قرار 8)
   736→   736→      fx = roundTo(toAmount.times(dec(rateDst)).minus(amountBase), 4);
   737→   737→    }
   738→   738→
   739→   739→    // 6) الإدراج
   740→   740→    const uid = getCurrentUserId() ?? null;
   741→   741→    const amountStr = money(amount);
   742→   742→    let sourceId: number;
   743→   743→
   744→   744→    if (v.txType === 'box_transfer') {
   745→   745→      const destId = v.toCashboxId as number;
   746→   746→      // الساق الصادرة من المصدر بعملته
   747→   747→      const res = await db.run(
   748→   748→        'INSERT INTO cash_tx(tx_type, cashbox_id, to_cashbox_id, currency_id, amount, exchange_rate, tx_date, ' +
   749→   749→          'ref_type, ref_id, customer_id, supplier_id, expense_category_id, description, created_at, created_by) ' +
   750→   750→          "VALUES('box_transfer', ?, ?, ?, ?, ?, ?, 'transfer', NULL, NULL, NULL, NULL, ?, ?, ?)",
   751→   751→        [v.cashboxId, destId, v.currencyId, amountStr, rateSrc.rate, txDate, description, now, uid],
   752→   752→      );
   753→   753→      sourceId = Number(res.lastInsertRowId);
   754→   754→      await db.run('UPDATE cash_tx SET ref_id = ? WHERE id = ?', [sourceId, sourceId]);
   755→   755→
   756→   756→      if (toAmount !== null && rateDst !== null && Number(destBox!.currency_id) !== v.currencyId) {
   757→   757→        // الساق الواردة إلى الوجهة بعملتها + settlement_rate + fx_gain_loss (قرار 8)
   758→   758→        await db.run(
   759→   759→          'INSERT INTO cash_tx(tx_type, cashbox_id, to_cashbox_id, currency_id, amount, exchange_rate, settlement_rate, ' +
   760→   760→            'fx_gain_loss, tx_date, ref_type, ref_id, customer_id, supplier_id, description, created_at, created_by) ' +
   761→   761→          "VALUES('box_transfer', ?, ?, ?, ?, ?, ?, ?, ?, 'transfer', ?, NULL, NULL, ?, ?, ?)",
   762→   762→          [
   763→   763→            v.cashboxId,
   764→   764→            destId,
   765→   765→            Number(destBox!.currency_id),
   766→   766→            money(toAmount),
   767→   767→            rateDst,
   768→   768→            rateDst,
   769→   769→            money(fx),
   770→   770→            txDate,
   771→   771→            sourceId,
   772→   772→            description,
   773→   773→            now,
   774→   774→            uid,
   775→   775→          ],
   776→   776→        );
   777→   777→      }
   778→   778→    } else {
   779→   779→      const res = await db.run(
   780→   780→        'INSERT INTO cash_tx(tx_type, cashbox_id, to_cashbox_id, currency_id, amount, exchange_rate, tx_date, ' +
   781→   781→          'ref_type, ref_id, expense_category_id, customer_id, supplier_id, description, created_at, created_by) ' +
   782→   782→          'VALUES(?, ?, NULL, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)',
   783→   783→        [
   784→   784→          v.txType,
   785→   785→          v.cashboxId,
   786→   786→          v.currencyId,
   787→   787→          amountStr,
   788→   788→          rateSrc.rate,
   789→   789→          txDate,
   790→   790→          refType,
   791→   791→          refId,
   792→   792→          expenseCategoryId,
   793→   793→          customerId,
   794→   794→          supplierId,
   795→   795→          description,
   796→   796→          now,
   797→   797→          uid,
   798→   798→        ],
   799→   799→      );
   800→   800→      sourceId = Number(res.lastInsertRowId);
   801→   801→    }
   802→   802→
   803→   803→    // 7) تخصيص FIFO للقبض الحر (نفس المعاملة — قاعدة 5.4-6)
   804→   804→    if (v.txType === 'receipt' && customerId !== null && refType === 'on_account') {
   805→   805→      await allocateReceiptFifo(db, sourceId, customerId, amount, now);
   806→   806→    }
   807→   807→
   808→   808→    // 7-ب) القبض/الصرف على فاتورة محددة: تخصيص مباشر بنفس النمط الذي تتبعه
   809→   809→    // لحظة البيع (applyCashIn) — يربط السند بفاتورته لأعمار الديون والتقارير
   810→   810→    if ((v.txType === 'receipt' || v.txType === 'payment') && refType === 'invoice' && refId !== null) {
   811→   811→      await db.run(
   812→   812→        'INSERT INTO payment_allocation(cash_tx_id, invoice_id, allocated_amount, allocated_at, created_by) VALUES(?, ?, ?, ?, ?)',
   813→   813→        [sourceId, refId, amountStr, now, uid],
   814→   814→      );
   815→   815→    }
   816→   816→
   817→   817→    // 8) الرصيد الجديد (قرار 9: بلا فحص — للتحذير في الواجهة)
   818→   818→    const flows = await sumBoxFlows(db, v.cashboxId, {});
   819→   819→    const newBalance = money(flows.in.minus(flows.out));
   820→   820→
   821→   821→    // 9) تدقيق الحوافز (FR-12-04 يوصي بـ owner_draw تحديداً)
   822→   822→    if (v.txType === 'owner_draw') {
   823→   823→      await logAudit('owner_draw', {
   824→   824→        entity: 'cash_tx',
   825→   825→        entityId: sourceId,
   826→   826→        details: { amount: amountStr, cashboxId: v.cashboxId, description },
   827→   827→      });
   828→   828→    }
   829→   829→
   830→   830→    return { id: sourceId, voucherNo: null, newBalance };
   831→   831→  });
   832→   832→}

  1020→  1020→    const fx = dec(r.fx_gain_loss);

  1022→  1022→      r.tx_type === 'box_transfer' ? 'transfer' : IN_TYPES.has(r.tx_type) ? 'in' : 'out';

  1121→  1121→ * بعملة الصندوق: موجبة لـ (receipt/capital_in/bank_deposit/opening أو تحويل وارد إلى الصندوق)

  1123→  1123→ * (دقة tx_date يومية — النافذة من تاريخ الفتح حتى اليوم.)
  1124→  1124→ */
  1125→  1125→export async function shiftExpected(cashboxId: number, openedAt: string): Promise<ShiftSummary> {
  1126→  1126→  const db = await getDb();
  1127→  1127→  const flows = await sumBoxFlows(db, cashboxId, { dateFrom: openedAt.slice(0, 10), dateTo: todayISO() });
  1128→  1128→  return {
  1129→  1129→    cashboxId,
  1130→  1130→    openedAt,
  1131→  1131→    expectedIn: money(flows.in),

  1133→  1133→    expected: money(flows.in.minus(flows.out)),
  1134→  1134→  };
  1135→  1135→}
  1136→  1136→
  1137→  1137→/**
  1138→  1138→ * إقفال الوردية: يسجل المتوقع (وارد − صادر خلال النافذة) والعد الفعلي والفرق
  1139→  1139→ * (الفرق = العد الفعلي − [الرصيد الافتتاحي + المتوقع] — العد يشمل الدرج كاملاً)
  1140→  1140→ * + قيد تدقيق. يعيد { expected, difference }.
  1141→  1141→ */
  1142→  1142→export async function closeShift(shiftId: number, counted: string, notes?: string): Promise<{ expected: string; difference: string }> {
  1143→  1143→  if (!/^\d+(\.\d+)?$/.test(counted ?? '')) {
  1144→  1144→    throw new Error('العد الفعلي يجب أن يكون رقماً غير سالب مثل «14500»');
  1145→  1145→  }
  1146→  1146→  const db = await getDb();
  1147→  1147→  return runSerialized(() =>
  1148→  1148→    db.transaction(async () => {
  1149→  1149→      const rows = await db.all<{
  1150→  1150→        id: number; cashbox_id: number; opened_at: string; closed_at: string | null; opening_count: string | number | null;
  1151→  1151→      }>('SELECT id, cashbox_id, opened_at, closed_at, opening_count FROM shift WHERE id = ?', [shiftId]);
  1152→  1152→      const sh = rows[0];
  1153→  1153→      if (sh === undefined) {
  1154→  1154→        throw new Error(`وردية غير موجودة (رقم ${shiftId}) — لا يمكن إقفالها`);
  1155→  1155→      }
  1156→  1156→      if (sh.closed_at !== null) {
  1157→  1157→        throw new Error('هذه الوردية مقفلة مسبقاً — لا يمكن إقفالها مرتين');
  1158→  1158→      }
  1159→  1159→
  1160→  1160→      const summary = await shiftExpected(Number(sh.cashbox_id), String(sh.opened_at));
  1161→  1161→      const countedD = dec(counted);
  1162→  1162→      const drawerExpected = dec(sh.opening_count ?? 0).plus(dec(summary.expected));
  1163→  1163→      const difference = roundTo(countedD.minus(drawerExpected), 4);
  1164→  1164→
  1165→  1165→      const now = new Date().toISOString();
  1166→  1166→      await db.run(
  1167→  1167→        'UPDATE shift SET closed_at = ?, expected = ?, counted = ?, difference = ?, notes = ?, updated_at = ? WHERE id = ?',
  1168→  1168→        [now, summary.expected, money(countedD), money(difference), notes?.trim() || null, now, shiftId],
  1169→  1169→      );
  1170→  1170→      await logAudit('shift_close', {
  1171→  1171→        entity: 'shift',
  1172→  1172→        entityId: shiftId,
  1173→  1173→        details: {
  1174→  1174→          cashboxId: Number(sh.cashbox_id),
  1175→  1175→          expected: summary.expected,
  1176→  1176→          counted: money(countedD),
  1177→  1177→          difference: money(difference),
  1178→  1178→        },
  1179→  1179→      });
  1180→  1180→      return { expected: summary.expected, difference: money(difference) };
  1181→  1181→    }),
  1182→  1182→  );
  1183→  1183→}
