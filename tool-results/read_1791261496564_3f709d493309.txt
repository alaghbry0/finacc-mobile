   289→/**
   290→ * Σ الوارد والصادر لصندوق بعملته خلال نافذة اختيارية — كل الجمع Decimal في JS
   291→ * (SUM في SQLite يُرجع REAL وقد يفقد الدقة). المستبعد دائماً: is_voided=1
   292→ * أو reversal_of NOT NULL (اصطلاح الإلغاء أعلاه).
   293→ */
   294→async function sumBoxFlows(
   295→  db: DbEngine,
   296→  cashboxId: number,
   297→  filter: { dateFrom?: string; dateTo?: string } = {},
   298→): Promise<{ in: Decimal; out: Decimal }> {
   299→  const boxRows = await db.all<{ currency_id: number }>('SELECT currency_id FROM cashbox WHERE id = ?', [cashboxId]);
   300→  if (boxRows.length === 0) {
   301→    throw new Error(`صندوق غير موجود (رقم ${cashboxId}) — لا يمكن حساب رصيده`);
   302→  }
   303→  const currencyId = Number(boxRows[0].currency_id);
   304→
   305→  const where: string[] = [
   306→    't.is_voided = 0',
   307→    't.reversal_of IS NULL',
   308→    't.currency_id = ?',
   309→    '(t.cashbox_id = ? OR (t.tx_type = \'box_transfer\' AND t.to_cashbox_id = ?))',
   310→  ];
   311→  const params: unknown[] = [currencyId, cashboxId, cashboxId];
   312→  if (filter.dateFrom !== undefined) {
   313→    where.push('t.tx_date >= ?');
   314→    params.push(filter.dateFrom);
   315→  }
   316→  if (filter.dateTo !== undefined) {
   317→    where.push('t.tx_date <= ?');
   318→    params.push(filter.dateTo);
   319→  }
   320→
   321→  const rows = await db.all<{ tx_type: string; cashbox_id: number; amount: string | number }>(
   322→    `SELECT t.tx_type, t.cashbox_id, t.amount FROM cash_tx t WHERE ${where.join(' AND ')}`,
   323→    params,
   324→  );
   325→
   326→  let inSum = dec(0);
   327→  let outSum = dec(0);
   328→  for (const r of rows) {
   329→    const amount = dec(r.amount);
   330→    if (r.tx_type === 'box_transfer') {
   331→      // ساق صادرة (cashbox_id=هذا) أو واردة (to_cashbox_id=هذا) — بعملة هذا الصندوق حصراً
   332→      if (Number(r.cashbox_id) === cashboxId) outSum = outSum.plus(amount);
   333→      else inSum = inSum.plus(amount);
   334→    } else if (IN_TYPES.has(r.tx_type)) {
   335→      inSum = inSum.plus(amount);
   336→    } else if (OUT_TYPES.has(r.tx_type)) {
   337→      outSum = outSum.plus(amount);
   338→    }
   339→  }
   340→  return { in: inSum, out: outSum };
   341→}

   345→/** أرصدة كل الصناديق الفاعلة بعملاتها — السالب مسموح ويعود كما هو (قرار 9). */
   346→export async function cashboxBalances(): Promise<CashboxBalanceRow[]> {
   347→  const db = await getDb();
   348→  const boxes = await db.all<{
   349→    id: number; name: string; currency_id: number; is_default: number; code: string; decimals: number;
   350→  }>(
   351→    'SELECT cb.id, cb.name, cb.currency_id, cb.is_default, c.code, c.decimals ' +
   352→      'FROM cashbox cb JOIN currency c ON c.id = cb.currency_id WHERE cb.is_archived = 0 ' +
   353→      'ORDER BY cb.is_default DESC, cb.name COLLATE NOCASE ASC',
   354→  );
   355→  const out: CashboxBalanceRow[] = [];
   356→  for (const b of boxes) {
   357→    const flows = await sumBoxFlows(db, Number(b.id), {});
   358→    out.push({
   359→      id: Number(b.id),
   360→      name: b.name,
   361→      currencyId: Number(b.currency_id),
   362→      currencyCode: b.code,
   363→      currencyDecimals: Number(b.decimals),
   364→      isDefault: Number(b.is_default) === 1,
   365→      balance: money(flows.in.minus(flows.out)),
   366→    });
   367→  }
   368→  return out;
   369→}

   396→// ============ المتبقي على فاتورة + FIFO (قاعدة 5.4-6) ============
   397→
   398→/**
   399→ * المتبقي على فاتورة = due_amount − Σ تخصيصات سنداتها التي تحمل الطرف
   400→ * (customer_id لفواتير البيع / supplier_id لفواتير الشراء) وغير الملغاة.
   401→ * سند لحظة البيع النقدي لا يحمل الطرف (قرار invoicing) فلا يُخصم مرتين.
   402→ */
   403→async function invoiceRemainingDue(
   404→  db: DbEngine,
   405→  invoiceId: number,
   406→  partyCol: 'customer_id' | 'supplier_id',
   407→): Promise<Decimal> {
   408→  const inv = (await db.all<{ due_amount: string | number }>('SELECT due_amount FROM invoice WHERE id = ?', [invoiceId]))[0];
   409→  if (inv === undefined) {
   410→    throw new Error(`فاتورة غير موجودة (رقم ${invoiceId}) — ربما حُذف الرابط، أعد اختيار الفاتورة`);
   411→  }
   412→  const allocs = await db.all<{ a: string | number }>(
   413→    `SELECT pa.allocated_amount AS a FROM payment_allocation pa JOIN cash_tx t ON t.id = pa.cash_tx_id ` +
   414→      `WHERE pa.invoice_id = ? AND t.${partyCol} IS NOT NULL AND t.is_voided = 0`,
   415→    [invoiceId],
   416→  );
   417→  let allocated = dec(0);
   418→  for (const r of allocs) allocated = allocated.plus(dec(r.a));
   419→  return dec(inv.due_amount).minus(allocated);
   420→}

   422→/**
   423→ * تخصيص مبلغ قبض على الحساب FIFO: أقدم فواتير البيع المكتملة الآجلة ذات متبقٍ
   424→ * للعميل ← payment_allocation حتى نفاد المبلغ. البقايا تبقى غير مخصصة = رصيد دائن.
   425→ */
   426→async function allocateReceiptFifo(
   427→  db: DbEngine,
   428→  cashTxId: number,
   429→  customerId: number,
   430→  amount: Decimal,
   431→  now: string,
   432→): Promise<void> {
   433→  const existing = await db.all<{ a: string | number }>(
   434→    'SELECT allocated_amount AS a FROM payment_allocation WHERE cash_tx_id = ?',
   435→    [cashTxId],
   436→  );
   437→  let left = dec(amount);
   438→  for (const r of existing) left = left.minus(dec(r.a));
   439→  if (left.lessThanOrEqualTo(0)) return;
   440→
   441→  const invoices = await db.all<{ id: number; due_amount: string | number }>(
   442→    "SELECT id, due_amount FROM invoice WHERE customer_id = ? AND doc_type = 'sale' AND status = 'completed' " +
   443→      "AND pay_status IN ('credit','mixed') AND due_amount > 0 ORDER BY issued_at ASC, id ASC",
   444→    [customerId],
   445→  );
   446→
   447→  for (const inv of invoices) {
   448→    if (left.lessThanOrEqualTo(0)) break;
   449→    const remaining = (await invoiceRemainingDue(db, Number(inv.id), 'customer_id')).toDecimalPlaces(4);
   450→    if (remaining.lessThanOrEqualTo(0)) continue;
   451→    const alloc = Decimal.min(remaining, left.toDecimalPlaces(4));
   452→    await db.run(
   453→      'INSERT INTO payment_allocation(cash_tx_id, invoice_id, allocated_amount, allocated_at, created_by) VALUES(?, ?, ?, ?, ?)',
   454→      [cashTxId, Number(inv.id), money(alloc), now, getCurrentUserId() ?? null],
   455→    );
   456→    left = left.minus(alloc);
   457→  }
   458→}

   460→/** تخصيص سند قبض حر (on_account) على الأقدم — يُستدعى من الواجهة عند إعادة التخصيص. */
   461→export async function allocateOnAccountReceipt(cashTxId: number): Promise<void> {
   462→  const db = await getDb();
   463→  await runSerialized(() =>
   464→    db.transaction(async () => {
   465→      const rows = await db.all<{
   466→        id: number; tx_type: string; customer_id: number | null; ref_type: string | null; amount: string | number; is_voided: number;
   467→      }>('SELECT id, tx_type, customer_id, ref_type, amount, is_voided FROM cash_tx WHERE id = ?', [cashTxId]);
   468→      const t = rows[0];
   469→      if (t === undefined) {
   470→        throw new Error(`حركة نقدية غير موجودة (رقم ${cashTxId}) — لا يمكن تخصيصها`);
   471→      }
   472→      if (Number(t.is_voided) === 1) {
   473→        throw new Error('هذه الحركة ملغاة — لا يمكن تخصيص قبض ملغى، أنشئ قبضاً جديداً');
   474→      }
   475→      if (t.tx_type !== 'receipt' || t.customer_id === null || t.ref_type !== 'on_account') {
   476→        throw new Error('التخصيص FIFO متاح لسندات القبض الحرّة (على الحساب) المرتبطة بعميل فقط');
   477→      }
   478→      await allocateReceiptFifo(db, cashTxId, Number(t.customer_id), dec(t.amount), new Date().toISOString());
   479→    }),
   480→  );
   481→}

   484→export async function customerOpenInvoices(customerId: number): Promise<OpenInvoiceRow[]> {
   485→  const db = await getDb();
   486→  const rows = await db.all<{ id: number; invoice_no: string | null; issued_at: string; due_amount: string | number; total: string | number }>(
   487→    "SELECT id, invoice_no, issued_at, due_amount, total FROM invoice WHERE customer_id = ? AND doc_type = 'sale' " +
   488→      "AND status = 'completed' AND pay_status IN ('credit','mixed') AND due_amount > 0 ORDER BY issued_at ASC, id ASC",
   489→    [customerId],
   490→  );
   491→  const out: OpenInvoiceRow[] = [];
   492→  for (const r of rows) {
   493→    const due = (await invoiceRemainingDue(db, Number(r.id), 'customer_id')).toDecimalPlaces(4);

   496→        id: Number(r.id),
   497→        invoiceNo: r.invoice_no,
   498→        issuedAt: r.issued_at,
   499→        total: money(r.total),
   500→        due: money(due),
   501→      });
   502→    }
   503→  }
   504→  return out;
   505→}

   543→      }
   544→      const destRows = await db.all<{ id: number; name: string; currency_id: number; is_archived: number; decimals: number }>(
   545→        'SELECT cb.id, cb.name, cb.currency_id, cb.is_archived, c.decimals FROM cashbox cb ' +
   546→          'JOIN currency c ON c.id = cb.currency_id WHERE cb.id = ?',

   548→      );

   551→        throw new Error(`صندوق الوجهة غير موجود (رقم ${v.toCashboxId}) — أعد اختيار الصندوق الوجهة`);
   552→      }
   553→      if (Number(destBox.is_archived) === 1) {
   554→        throw new Error(`صندوق الوجهة «${destBox.name}» مؤرشف — اختر صندوقاً فاعلاً`);
   555→      }
   556→    }
   557→
   558→    // 2) الفترات والتأريخ (قاعدة 5.4-11)
   559→    await assertPeriodOpen(txDate);
   560→    await assertBackdateAllowed(txDate, { managerConfirmed: v.managerConfirmedBackdate === true });
   561→
   562→    // 3) سعر الصرف لحظة الحركة (قرار 3: MissingRateError تمر كما هي قبل أي كتابة)

   567→    let customerId = v.customerId ?? null;
   568→    let supplierId = v.supplierId ?? null;

   570→    let refId = v.refId ?? null;

   572→    const description = v.description?.trim() || null;
   573→
   574→    if (v.txType === 'expense') {
   575→      if (expenseCategoryId === null) {
   576→        throw new Error('المصروف يتطلب فئة مصروف — اختر الفئة من القائمة أو أنشئ فئة جديدة');

   579→        'SELECT id, name, is_archived FROM expense_category WHERE id = ?',
   580→        [expenseCategoryId],
   581→      );
   582→      const cat = catRows[0];
   583→      if (cat === undefined) {
   584→        throw new Error(`فئة مصروف غير موجودة (رقم ${expenseCategoryId}) — أعد اختيار الفئة`);

   595→        throw new Error('سند القبض يخص تحصيلاً من عميل — أزل المورّد أو استخدم «صرف دفع لمورّد»');

   631→            throw new Error('الفاتورة المحددة نقديّة مسددة بالكامل — لا يمكن تحصيلها مرتين');

   637→          if (amount.greaterThan(remaining)) {
   638→            throw new Error(
   639→              `المبلغ يتجاوز المتبقي على الفاتورة (المتبقي ${money(remaining)}) — ` +
   640→                'خفّض المبلغ أو استخدم «على الحساب» ليُخصم الفائض من الفواتير الأقدم أولاً',

   642→          }
   643→        }
   644→      } else if (refType === 'on_account' || refType === 'invoice') {

   656→          [supplierId],

   678→            inv === undefined ||
   679→            inv.doc_type !== 'purchase' ||

   708→      supplierId = null;
   709→      refType = null;
   710→      refId = null;

   713→    // 5) حسابات التحويل بين عملتين (قرار 8 / FR-04-07)

   715→    let rateDst: string | null = null;

   727→      const amountBase = amount.times(dec(rateSrc.rate)); // قيمة المبلغ بالعملة الأساسية

   729→        v.toAmount !== undefined

   740→    const uid = getCurrentUserId() ?? null;
   741→    const amountStr = money(amount);

   761→          "VALUES('box_transfer', ?, ?, ?, ?, ?, ?, ?, ?, 'transfer', ?, NULL, NULL, ?, ?, ?)",

   763→            v.cashboxId,
   764→            destId,
   765→            Number(destBox!.currency_id),
   766→            money(toAmount),
   767→            rateDst,
   768→            rateDst,
   769→            money(fx),
   770→            txDate,
   771→            sourceId,
   772→            description,
   773→            now,
   774→            uid,
   775→          ],
   776→        );
   777→      }
   778→    } else {
   779→      const res = await db.run(
   780→        'INSERT INTO cash_tx(tx_type, cashbox_id, to_cashbox_id, currency_id, amount, exchange_rate, tx_date, ' +
   781→          'ref_type, ref_id, expense_category_id, customer_id, supplier_id, description, created_at, created_by) ' +
   782→          'VALUES(?, ?, NULL, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)',
   783→        [
   784→          v.txType,
   785→          v.cashboxId,
   786→          v.currencyId,
   787→          amountStr,
   788→          rateSrc.rate,
   789→          txDate,
   790→          refType,
   791→          refId,
   792→          expenseCategoryId,
   793→          customerId,
   794→          supplierId,
   795→          description,
   796→          now,
   797→          uid,
   798→        ],
   799→      );
   800→      sourceId = Number(res.lastInsertRowId);
   801→    }
   802→
   803→    // 7) تخصيص FIFO للقبض الحر (نفس المعاملة — قاعدة 5.4-6)
   804→    if (v.txType === 'receipt' && customerId !== null && refType === 'on_account') {
   805→      await allocateReceiptFifo(db, sourceId, customerId, amount, now);
   806→    }
   807→
   808→    // 7-ب) القبض/الصرف على فاتورة محددة: تخصيص مباشر بنفس النمط الذي تتبعه
   809→    // لحظة البيع (applyCashIn) — يربط السند بفاتورته لأعمار الديون والتقارير
   810→    if ((v.txType === 'receipt' || v.txType === 'payment') && refType === 'invoice' && refId !== null) {
   811→      await db.run(
   812→        'INSERT INTO payment_allocation(cash_tx_id, invoice_id, allocated_amount, allocated_at, created_by) VALUES(?, ?, ?, ?, ?)',
   813→        [sourceId, refId, amountStr, now, uid],
   814→      );
   815→    }
   816→
   817→    // 8) الرصيد الجديد (قرار 9: بلا فحص — للتحذير في الواجهة)
   818→    const flows = await sumBoxFlows(db, v.cashboxId, {});
   819→    const newBalance = money(flows.in.minus(flows.out));
   820→
   821→    // 9) تدقيق الحوافز (FR-12-04 يوصي بـ owner_draw تحديداً)
   822→    if (v.txType === 'owner_draw') {
   823→      await logAudit('owner_draw', {
   824→        entity: 'cash_tx',
   825→        entityId: sourceId,
   826→        details: { amount: amountStr, cashboxId: v.cashboxId, description },
   827→      });
   828→    }
   829→
   830→    return { id: sourceId, voucherNo: null, newBalance };
   831→  });
   832→}

   885→          swapBoxes ? Number(leg.to_cashbox_id ?? leg.cashbox_id) : Number(leg.cashbox_id),
   886→          swapBoxes ? Number(leg.cashbox_id) : (leg.to_cashbox_id === null ? null : Number(leg.to_cashbox_id)),
   887→          Number(leg.currency_id),
   888→          money(leg.amount),

  1121→ * بعملة الصندوق: موجبة لـ (receipt/capital_in/bank_deposit/opening أو تحويل وارد إلى الصندوق)
  1122→ * وسالبة لـ (payment/expense/owner_draw/bank_withdraw أو تحويل صادر من الصندوق).
  1123→ * (دقة tx_date يومية — النافذة من تاريخ الفتح حتى اليوم.)

  1125→export async function shiftExpected(cashboxId: number, openedAt: string): Promise<ShiftSummary> {
  1126→  const db = await getDb();
  1127→  const flows = await sumBoxFlows(db, cashboxId, { dateFrom: openedAt.slice(0, 10), dateTo: todayISO() });
  1128→  return {
  1129→    cashboxId,
  1130→    openedAt,
  1131→    expectedIn: money(flows.in),
  1132→    expectedOut: money(flows.out),
  1133→    expected: money(flows.in.minus(flows.out)),
  1134→  };
  1135→}
  1136→
  1137→/**
  1138→ * إقفال الوردية: يسجل المتوقع (وارد − صادر خلال النافذة) والعد الفعلي والفرق
  1139→ * (الفرق = العد الفعلي − [الرصيد الافتتاحي + المتوقع] — العد يشمل الدرج كاملاً)
  1140→ * + قيد تدقيق. يعيد { expected, difference }.
  1141→ */
  1142→export async function closeShift(shiftId: number, counted: string, notes?: string): Promise<{ expected: string; difference: string }> {
  1143→  if (!/^\d+(\.\d+)?$/.test(counted ?? '')) {
  1144→    throw new Error('العد الفعلي يجب أن يكون رقماً غير سالب مثل «14500»');
  1145→  }
  1146→  const db = await getDb();
  1147→  return runSerialized(() =>
  1148→    db.transaction(async () => {
  1149→      const rows = await db.all<{
  1150→        id: number; cashbox_id: number; opened_at: string; closed_at: string | null; opening_count: string | number | null;
  1151→      }>('SELECT id, cashbox_id, opened_at, closed_at, opening_count FROM shift WHERE id = ?', [shiftId]);
  1152→      const sh = rows[0];
  1153→      if (sh === undefined) {
  1154→        throw new Error(`وردية غير موجودة (رقم ${shiftId}) — لا يمكن إقفالها`);
  1155→      }
  1156→      if (sh.closed_at !== null) {
  1157→        throw new Error('هذه الوردية مقفلة مسبقاً — لا يمكن إقفالها مرتين');
  1158→      }
  1159→
  1160→      const summary = await shiftExpected(Number(sh.cashbox_id), String(sh.opened_at));
  1161→      const countedD = dec(counted);
  1162→      const drawerExpected = dec(sh.opening_count ?? 0).plus(dec(summary.expected));
  1163→      const difference = roundTo(countedD.minus(drawerExpected), 4);
  1164→
  1165→      const now = new Date().toISOString();
  1166→      await db.run(
  1167→        'UPDATE shift SET closed_at = ?, expected = ?, counted = ?, difference = ?, notes = ?, updated_at = ? WHERE id = ?',
  1168→        [now, summary.expected, money(countedD), money(difference), notes?.trim() || null, now, shiftId],
  1169→      );
  1170→      await logAudit('shift_close', {
  1171→        entity: 'shift',
  1172→        entityId: shiftId,
  1173→        details: {
  1174→          cashboxId: Number(sh.cashbox_id),
  1175→          expected: summary.expected,
  1176→          counted: money(countedD),
  1177→          difference: money(difference),
  1178→        },
  1179→      });
  1180→      return { expected: summary.expected, difference: money(difference) };
  1181→    }),
  1182→  );

  1217→export async function upsertExpenseCategory(id: number | null, name: string): Promise<number> {
  1218→  const clean = (name ?? '').trim();
  1219→  if (clean.length === 0) {
  1220→    throw new Error('اسم فئة المصروف مطلوب — أدخل اسماً مثل «نظافة»');
  1221→  }
  1222→  if (clean.length > 60) {
  1223→    throw new Error('اسم فئة المصروف طويل (الحد 60 حرفاً)');
  1224→  }
  1225→  const db = await getDb();
  1226→  const dupRows = await db.all<{ id: number }>(
  1227→    'SELECT id FROM expense_category WHERE is_archived = 0 AND name = ? COLLATE NOCASE',
  1228→    [clean],
  1229→  );
  1230→  const dup = dupRows.find((r) => id === null || Number(r.id) !== id);
  1231→  if (dup !== undefined) {
  1232→    throw new Error(`فئة المصروف «${clean}» موجودة مسبقاً — استخدم الفئة القائمة`);
  1233→  }
  1234→  if (id === null) {
  1235→    const res = await db.run('INSERT INTO expense_category(name, is_archived, created_at, created_by) VALUES(?, 0, ?, ?)', [
  1236→      clean,
  1237→      new Date().toISOString(),
  1238→      getCurrentUserId() ?? null,
  1239→    ]);
  1240→    return Number(res.lastInsertRowId);
  1241→  }
  1242→  const exists = await db.all<{ id: number }>('SELECT id FROM expense_category WHERE id = ?', [id]);
  1243→  if (exists.length === 0) {
  1244→    throw new Error(`فئة مصروف غير موجودة (رقم ${id}) — لا يمكن تعديلها`);
  1245→  }
  1246→  await db.run('UPDATE expense_category SET name = ?, updated_at = ? WHERE id = ?', [
  1247→    clean,
  1248→    new Date().toISOString(),
  1249→    id,
  1250→  ]);
  1251→  return id;
  1252→}
